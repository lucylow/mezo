## 1. Project Architecture Overview

The frontend for the Auto‑Compounder Vault is built as a **Next.js 14 App Router** application with **TypeScript** for type safety. It integrates with the Mezo blockchain via **wagmi** and **RainbowKit**, extended by **Mezo Passport** to support both Bitcoin wallets (Xverse, Unisat) and EVM wallets (MetaMask).

**Core Design Principles:**
- **Separation of Concerns**: Hooks for data fetching, components for presentation, stores for global state.
- **Optimistic UI**: Immediate feedback on transactions with rollback on failure.
- **Responsive Design**: Mobile‑first approach using Tailwind CSS breakpoints.
- **Accessibility**: WCAG 2.2 AA compliance with keyboard navigation and screen reader support.
- **Performance**: Static generation where possible, lazy loading for heavy components.

**Technology Stack:**

| Category | Technology | Purpose |
|:---|:---|:---|
| Framework | Next.js 14 (App Router) | Server‑side rendering, routing |
| Language | TypeScript | Type safety |
| Styling | Tailwind CSS | Utility‑first styling |
| Web3 | wagmi, viem, RainbowKit | Blockchain interactions |
| Mezo Integration | @mezo‑org/passport | Bitcoin + EVM wallet support |
| State Management | Zustand | Global UI and user state |
| Data Fetching | TanStack Query | Server‑state management |
| Charts | Recharts | Data visualization |
| Animations | Framer Motion | Smooth transitions |
| Notifications | Sonner | Toast notifications |

```tsx
'use client';

import '@rainbow-me/rainbowkit/styles.css';

// Custom theme for RainbowKit to match Mezo branding
import { darkTheme } from '@rainbow-me/rainbowkit';

const customTheme = {
  ...darkTheme(),
  colors: {
    ...darkTheme().colors,
    accentColor: '#F5A623',
    accentColorForeground: '#FFFFFF',
    connectButtonBackground: 'rgba(20, 22, 26, 0.8)',
    connectButtonBackgroundError: '#EF4444',
    connectButtonText: '#FFFFFF',
    modalBackground: '#0A0B0D',
    modalBorder: 'rgba(255, 255, 255, 0.08)',
  },
  radii: {
    ...darkTheme().radii,
    connectButton: '12px',
    modal: '16px',
    menuButton: '12px',
  },
};

export function Providers({ children }: { children: React.ReactNode }) {
  const [queryClient] = useState(
    () =>
      new QueryClient({
        defaultOptions: {
          queries: {
            staleTime: 30_000, // 30 seconds
            refetchOnWindowFocus: false,
            retry: 1,
          },
        },
      })
  );

  const config = getDefaultConfig({
    appName: 'veMEZO Auto-Compounder',
    projectId: process.env.NEXT_PUBLIC_WALLETCONNECT_PROJECT_ID!,
    chains: [mezoMatsnetTestnet],
    ...getMezoPassportConfig(),
  });

  return (
    <WagmiProvider config={config}>
      <QueryClientProvider client={queryClient}>
        <RainbowKitProvider theme={customTheme} modalSize="compact">
          {children}
          <Toaster 
            position="top-right" 
            theme="dark"
            toastOptions={{
              style: {
                background: '#0A0B0D',
                border: '1px solid rgba(255, 255, 255, 0.08)',
                color: '#FFFFFF',
              },
            }}
          />
        </RainbowKitProvider>
      </QueryClientProvider>
    </WagmiProvider>
  );
}
```

## 5. Core Custom Hooks

### 5.1 `hooks/useVaultStats.ts`

```typescript
import { useReadContract, useReadContracts } from 'wagmi';
import { formatEther } from 'viem';
import { CONTRACTS } from '@/lib/contracts/addresses';
import { VeMEZOVaultABI } from '@/lib/contracts/abis/VeMEZOVault';

export function useVaultStats() {
  const { data: contractData, refetch } = useReadContracts({
    contracts: [
      {
        address: CONTRACTS.VAULT,
        abi: VeMEZOVaultABI,
        functionName: 'totalUnderlying',
      },
      {
        address: CONTRACTS.VAULT,
        abi: VeMEZOVaultABI,
        functionName: 'totalSupply',
      },
      {
        address: CONTRACTS.VAULT,
        abi: VeMEZOVaultABI,
        functionName: 'performanceFee',
      },
      {
        address: CONTRACTS.VAULT,
        abi: VeMEZOVaultABI,
        functionName: 'lastCompoundTime',
      },
      {
        address: CONTRACTS.VAULT,
        abi: VeMEZOVaultABI,
        functionName: 'getPendingRewards',
      },
    ],
  });

  const totalUnderlying = contractData?.[0]?.result as bigint | undefined;
  const totalSupply = contractData?.[1]?.result as bigint | undefined;
  const performanceFee = contractData?.[2]?.result as bigint | undefined;
  const lastCompoundTime = contractData?.[3]?.result as bigint | undefined;
  const pendingRewards = contractData?.[4]?.result as bigint | undefined;

  const tvl = totalUnderlying ? Number(formatEther(totalUnderlying)) : 0;
  const feePercentage = performanceFee ? Number(performanceFee) / 100 : 0;
  const pending = pendingRewards ? Number(formatEther(pendingRewards)) : 0;

  // Calculate projected APR based on TVL and recent compounding
  const projectedAPR = calculateProjectedAPR(tvl, pending);

  // Calculate time since last compound
  const lastCompound = lastCompoundTime ? new Date(Number(lastCompoundTime) * 1000) : null;
  const timeSinceCompound = lastCompound 
    ? Math.floor((Date.now() - lastCompound.getTime()) / (1000 * 60 * 60))
    : 0;

  return {
    tvl,
    totalSupply: totalSupply ? formatEther(totalSupply) : '0',
    performanceFee: feePercentage,
    pendingRewards: pending,
    projectedAPR,
    lastCompoundTime: lastCompound,
    timeSinceCompound,
    isLoading: !contractData,
    refetch,
  };
}

function calculateProjectedAPR(tvl: number, pendingRewards: number): number {
  // Base APR from protocol yields (50% average from Mezo Earn)
  const baseAPR = 50;
  
  // Boost from vault efficiency (more TVL = better aggregation = higher yields)
  const efficiencyBoost = Math.min(30, Math.floor(tvl / 50000));
  
  // Recent reward rate indicator
  const rewardBonus = pendingRewards > 100 ? 10 : 0;
  
  return baseAPR + efficiencyBoost + rewardBonus;
}
```

### 5.2 `hooks/useUserPosition.ts`

```typescript
import { useAccount, useReadContracts } from 'wagmi';
import { formatEther } from 'viem';
import { CONTRACTS } from '@/lib/contracts/addresses';
import { VeMEZOVaultABI } from '@/lib/contracts/abis/VeMEZOVault';
import { VeMEZOABI } from '@/lib/contracts/abis/VeMEZO';

export function useUserPosition() {
  const { address, isConnected } = useAccount();

  const { data, refetch, isLoading } = useReadContracts({
    contracts: [
      {
        address: CONTRACTS.VAULT,
        abi: VeMEZOVaultABI,
        functionName: 'balanceOf',
        args: [address!],
      },
      {
        address: CONTRACTS.VAULT,
        abi: VeMEZOVaultABI,
        functionName: 'getUserTokenIds',
        args: [address!],
      },
      {
        address: CONTRACTS.VEMEZO,
        abi: VeMEZOABI,
        functionName: 'tokensOfOwner',
        args: [address!],
      },
    ],
    query: {
      enabled: isConnected && !!address,
    },
  });

  const vaultShares = data?.[0]?.result as bigint | undefined;
  const depositedTokenIds = data?.[1]?.result as bigint[] | undefined;
  const ownedVeMEZOTokens = data?.[2]?.result as bigint[] | undefined;

  // Fetch total supply for share calculation
  const { data: totalSupply } = useReadContract({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    functionName: 'totalSupply',
  });

  const sharePercentage = vaultShares && totalSupply
    ? (Number(formatEther(vaultShares)) / Number(formatEther(totalSupply as bigint))) * 100
    : 0;

  // Fetch underlying value for user's share
  const { data: totalUnderlying } = useReadContract({
    address: CONTRACTS.VAULT,
```

```typescript
  value: bigint;
  lockEnd: Date;
  lockAmount: bigint;

export function useVeMEZONFTs() {
  const { address, isConnected } = useAccount();

  const { data: tokenIds } = useReadContract({
    address: CONTRACTS.VEMEZO,
    abi: VeMEZOABI,
    functionName: 'tokensOfOwner',
    args: [address!],
    query: {
      enabled: isConnected && !!address,
    },
  });

  const tokenIdsArray = (tokenIds as bigint[]) || [];

  const { data: nftDetails } = useReadContracts({
    contracts: tokenIdsArray.flatMap((tokenId) => [
      {
        address: CONTRACTS.VEMEZO,
        abi: VeMEZOABI,
        functionName: 'balanceOfNFT',
        args: [tokenId],
      },
      {
        address: CONTRACTS.VEMEZO,
        abi: VeMEZOABI,
        functionName: 'locked',
        args: [tokenId],
      },
    ]),
    query: {
      enabled: tokenIdsArray.length > 0,
    },
  });

  const nfts: VeMEZONFT[] = [];
  for (let i = 0; i < tokenIdsArray.length; i++) {

        value,
        lockAmount: locked[0],
        lockEnd: new Date(Number(locked[1]) * 1000),
      });
    }
  }

  return {
    nfts,
    isLoading: !nftDetails && tokenIdsArray.length > 0,
```

```tsx
'use client';

import { createContext, useContext, useState } from 'react';
import { cn } from '@/lib/utils';
import { motion } from 'framer-motion';

interface TabsContextType {
  activeTab: string;
  setActiveTab: (value: string) => void;
}

const TabsContext = createContext<TabsContextType | undefined>(undefined);

interface TabsProps {
  defaultValue: string;
  children: React.ReactNode;
  className?: string;
}

export function Tabs({ defaultValue, children, className }: TabsProps) {
  const [activeTab, setActiveTab] = useState(defaultValue);
  return (
    <TabsContext.Provider value={{ activeTab, setActiveTab }}>
      <div className={cn('', className)}>{children}</div>
    </TabsContext.Provider>
  );
}

interface TabsListProps {
  children: React.ReactNode;
  className?: string;
}

export function TabsList({ children, className }: TabsListProps) {
  return (
    <div className={cn('flex border-b border-border', className)}>
      {children}
    </div>
  );
}

interface TabsTriggerProps {
  value: string;
  children: React.ReactNode;
  className?: string;
}

export function TabsTrigger({ value, children, className }: TabsTriggerProps) {
  const context = useContext(TabsContext);
  if (!context) throw new Error('TabsTrigger must be used within Tabs');
  
  const isActive = context.activeTab === value;
  
  return (
    <button
      onClick={() => context.setActiveTab(value)}
      className={cn(
        'flex-1 py-3 font-medium transition relative',
        isActive ? 'text-mezo-primary' : 'text-mezo-muted hover:text-white',
        className
      )}
    >
      {children}
      {isActive && (
        <motion.div
          layoutId="activeTab"
          className="absolute bottom-0 left-0 right-0 h-0.5 bg-mezo-primary"
        />
      )}
    </button>
  );
}

interface TabsContentProps {
  value: string;
  children: React.ReactNode;
  className?: string;
}

export function TabsContent({ value, children, className }: TabsContentProps) {
  const context = useContext(TabsContext);
  if (!context) throw new Error('TabsContent must be used within Tabs');
  
  if (context.activeTab !== value) return null;
  
  return (
    <motion.div
      initial={{ opacity: 0, y: 10 }}
      animate={{ opacity: 1, y: 0 }}
      exit={{ opacity: 0, y: -10 }}
      className={cn('pt-4', className)}
    >
      {children}
    </motion.div>
  );
}
```

```tsx
'use client';

import { useState } from 'react';
import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/Card';
import { Slider } from '@/components/ui/Slider';
import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer } from 'recharts';

export function CompoundingSimulator() {
  const [deposit, setDeposit] = useState(1000);
  const [apy, setApy] = useState(80);
  const [years, setYears] = useState(2);

  const calculateGrowth = () => {
    const data = [];
    const periodsPerYear = 52;
    const ratePerPeriod = apy / 100 / periodsPerYear;
    let balance = deposit;
    
    for (let week = 0; week <= years * 52; week++) {
      if (week % 4 === 0) {
        data.push({
          month: week / 4,
          balance: Math.round(balance),
        });
      }
      balance = balance * (1 + ratePerPeriod);
    }
    return data;
  };

  const chartData = calculateGrowth();
  const finalBalance = chartData[chartData.length - 1]?.balance || 0;
  const totalEarned = finalBalance - deposit;

  return (
    <Card>
      <CardHeader>
        <CardTitle>Simulate Your Earnings</CardTitle>
```

## 12. Transaction Monitoring & Notifications

### 12.1 `components/TransactionToast.tsx`

```tsx
'use client';

import { useEffect } from 'react';
import { toast } from 'sonner';
import { useWaitForTransactionReceipt } from 'wagmi';
import { CheckCircle2, XCircle, Loader2 } from 'lucide-react';

interface TransactionToastProps {
  hash: `0x${string}`;
  onSuccess?: () => void;
  onError?: () => void;
}

export function useTransactionToast({ hash, onSuccess, onError }: TransactionToastProps) {
  const { isLoading, isSuccess, isError, error } = useWaitForTransactionReceipt({ hash });

  useEffect(() => {
    if (isLoading) {
      toast.loading('Transaction pending...', {
        id: hash,
        icon: <Loader2 className="h-4 w-4 animate-spin" />,
      });
    } else if (isSuccess) {
      toast.success('Transaction confirmed!', {
        id: hash,
        icon: <CheckCircle2 className="h-4 w-4 text-mezo-accent" />,
      });
      onSuccess?.();
    } else if (isError) {
      toast.error(`Transaction failed: ${error?.message}`, {
        id: hash,
        icon: <XCircle className="h-4 w-4 text-red-500" />,
      });
      onError?.();
    }
  }, [isLoading, isSuccess, isError, error, hash, onSuccess, onError]);

  return { isLoading, isSuccess, isError };
}
```

## 13. Theme & Styling System

```typescript
import { type ClassValue, clsx } from 'clsx';
import { twMerge } from 'tailwind-merge';

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}

export function formatNumber(num: number): string {
  if (num >= 1_000_000) {
    return (num / 1_000_000).toFixed(2) + 'M';
  }
  if (num >= 1_000) {
    return (num / 1_000).toFixed(2) + 'K';
  }
  return num.toFixed(2);
}

export function shortenAddress(address: string): string {
  return `${address.slice(0, 6)}...${address.slice(-4)}`;
}

export function formatDate(date: Date): string {
  return date.toLocaleDateString('en-US', {
    month: 'short',
    day: 'numeric',
    year: 'numeric',
  });
}

export function calculateTimeUntilEpoch(): string {
  const now = new Date();
  const nextThursday = new Date(now);
  nextThursday.setUTCDate(now.getUTCDate() + ((4 + 7 - now.getUTCDay()) % 7));
  nextThursday.setUTCHours(0, 0, 0, 0);
  
  const diff = nextThursday.getTime() - now.getTime();
  const days = Math.floor(diff / (1000 * 60 * 60 * 24));
  const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
  
  return `${days}d ${hours}h`;
}
```

## 14. Error Handling & Loading States

### 14.1 `components/ErrorBoundary.tsx`

```tsx
'use client';

import { Component, ErrorInfo, ReactNode } from 'react';
import { Button } from '@/components/ui/Button';
import { AlertTriangle } from 'lucide-react';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('ErrorBoundary caught:', error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      if (this.props.fallback) {
        return this.props.fallback;
      }
      
      return (
        <div className="flex flex-col items-center justify-center min-h-[400px] p-8">
          <AlertTriangle className="h-16 w-16 text-mezo-primary mb-4" />
          <h2 className="text-2xl font-bold mb-2">Something went wrong</h2>
          <p className="text-mezo-muted mb-6 text-center max-w-md">
            {this.state.error?.message || 'An unexpected error occurred.'}
          </p>
          <Button onClick={() => window.location.reload()}>
            Refresh Page
          </Button>
        </div>
      );
    }

    return this.props.children;
  }
}
```
