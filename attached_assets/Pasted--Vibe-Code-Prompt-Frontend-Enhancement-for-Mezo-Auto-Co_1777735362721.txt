### 2.2. WithdrawForm

Allows withdrawal by NFT or by shares.

```tsx
// components/vault/WithdrawForm.tsx
'use client';

import { useState } from 'react';
import { useWithdraw } from '@/hooks/useVaultWrite';
import { useUserPosition } from '@/hooks/useUserPosition';
import { Button } from '@/components/ui/Button';

export function WithdrawForm() {
  const { depositedTokenIds, shares } = useUserPosition();
  const { withdraw, withdrawByShares, isPending } = useWithdraw();
  const [mode, setMode] = useState<'token' | 'shares'>('token');
  const [selectedTokenId, setSelectedTokenId] = useState('');
  const [shareAmount, setShareAmount] = useState('');

  const handleWithdraw = async () => {
    if (mode === 'token' && selectedTokenId) {
      await withdraw(BigInt(selectedTokenId));
    } else if (mode === 'shares' && shareAmount) {
      await withdrawByShares(BigInt(shareAmount));
    }
  };

  return (
    <div className="space-y-4">
      <div className="flex gap-2">
        <Button variant={mode === 'token' ? 'primary' : 'outline'} size="sm" onClick={() => setMode('token')}>
          By NFT
        </Button>
        <Button variant={mode === 'shares' ? 'primary' : 'outline'} size="sm" onClick={() => setMode('shares')}>
          By Shares
        </Button>
      </div>

      {mode === 'token' ? (
        <select
          className="input-field"
          value={selectedTokenId}
          onChange={(e) => setSelectedTokenId(e.target.value)}
        >
          <option value="">Select deposited NFT...</option>
          {depositedTokenIds.map((id) => (
            <option key={id.toString()} value={id.toString()}>
              Token #{id.toString()}
            </option>
          ))}
        </select>
      ) : (
        <div>
          <input
            type="number"
            placeholder="Enter share amount"
            className="input-field"
            value={shareAmount}
            onChange={(e) => setShareAmount(e.target.value)}
          />
          <p className="text-sm text-mezo-muted mt-1">
            Available: {Number(shares).toFixed(2)} vveMEZO
          </p>
        </div>
      )}

      <Button onClick={handleWithdraw} disabled={isPending} fullWidth>
        {isPending ? 'Withdrawing...' : 'Withdraw'}
      </Button>
    </div>
  );
}
```

## 3. Compounding Simulator (Interactive Chart)

Lets users project their earnings over time with adjustable parameters.

```tsx
// components/sections/CompoundingSimulator.tsx
'use client';

import { useState } from 'react';
import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/Card';
import { Slider } from '@/components/ui/Slider';
import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer } from 'recharts';

export function CompoundingSimulator() {
  const [deposit, setDeposit] = useState(1000);
  const [apy, setApy] = useState(80);
  const [years, setYears] = useState(2);

  const calculateGrowth = () => {
    const periodsPerYear = 52; // weekly compounding
    const ratePerPeriod = apy / 100 / periodsPerYear;
    let balance = deposit;
    const data = [];
    for (let w = 0; w <= years * 52; w++) {
      if (w % 4 === 0) {
        data.push({ month: w / 4, balance: Math.round(balance) });
      }
      balance *= 1 + ratePerPeriod;
    }
    return data;
  };

  const chartData = calculateGrowth();
  const finalBalance = chartData[chartData.length - 1]?.balance || 0;

  return (
    <Card>
      <CardHeader>
        <CardTitle>Compounding Simulator</CardTitle>
      </CardHeader>
      <CardContent>
        <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
          <div className="space-y-6">
            <div>
              <label className="block text-sm font-medium mb-2">
                Initial Deposit: {deposit.toLocaleString()} MEZO
              </label>
              <Slider
                value={[deposit]}
                onValueChange={([v]) => setDeposit(v)}
                min={100}
                max={100000}
                step={100}
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-2">APY: {apy}%</label>
              <Slider value={[apy]} onValueChange={([v]) => setApy(v)} min={10} max={200} step={1} />
            </div>
            <div>
              <label className="block text-sm font-medium mb-2">Years: {years}</label>
              <Slider value={[years]} onValueChange={([v]) => setYears(v)} min={1} max={4} step={0.5} />
            </div>
            <div className="pt-4 border-t border-border">
              <p className="text-sm text-mezo-muted">Final Balance</p>
              <p className="text-3xl font-bold gradient-text">{finalBalance.toLocaleString()} MEZO</p>
            </div>
          </div>
          <div className="lg:col-span-2 h-72">
            <ResponsiveContainer>
              <LineChart data={chartData}>
                <CartesianGrid strokeDasharray="3 3" stroke="rgba(255,255,255,0.1)" />
                <XAxis dataKey="month" stroke="#6B7280" />
                <YAxis tickFormatter={(v) => `${(v / 1000).toFixed(0)}k`} stroke="#6B7280" />
                <Tooltip
                  contentStyle={{ backgroundColor: '#0A0B0D', border: '1px solid rgba(255,255,255,0.1)' }}
                />
                <Line type="monotone" dataKey="balance" stroke="#F5A623" strokeWidth={3} dot={false} />
              </LineChart>
            </ResponsiveContainer>
          </div>
        </div>
      </CardContent>
    </Card>
  );
}
```

## 4. Real‑Time Transaction History

Uses `useWatchContractEvent` to listen to vault events and display them in a feed.

```tsx
// components/sections/RecentActivity.tsx
'use client';

import { useWatchContractEvent } from 'wagmi';
import { useState } from 'react';
import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/Card';
import { ArrowDown, ArrowUp, RefreshCw } from 'lucide-react';
import { CONTRACTS, VeMEZOVaultABI } from '@/lib/contracts';
import { formatEther } from 'viem';

interface Activity {
  type: 'deposit' | 'withdraw' | 'compound';
  user: string;
  amount: string;
  timestamp: Date;
}

export function RecentActivity() {
  const [activities, setActivities] = useState<Activity[]>([]);

  useWatchContractEvent({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    eventName: 'Deposited',
    onLogs(logs) {
      const newActs = logs.map((log) => ({
        type: 'deposit' as const,
        user: log.args.user as string,
        amount: formatEther(log.args.value as bigint),
        timestamp: new Date(),
      }));
      setActivities((prev) => [...newActs, ...prev].slice(0, 10));
    },
  });

  useWatchContractEvent({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    eventName: 'Withdrawn',
    onLogs(logs) {
      const newActs = logs.map((log) => ({
        type: 'withdraw' as const,
        user: log.args.user as string,
        amount: formatEther(log.args.value as bigint),
        timestamp: new Date(),
      }));
      setActivities((prev) => [...newActs, ...prev].slice(0, 10));
    },
  });

  useWatchContractEvent({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    eventName: 'Compounded',
    onLogs(logs) {
      const newActs = logs.map((log) => ({
        type: 'compound' as const,
        user: 'Keeper',
        amount: formatEther(log.args.totalRewards as bigint),
        timestamp: new Date(),
      }));
      setActivities((prev) => [...newActs, ...prev].slice(0, 10));
    },
  });

  const getIcon = (type: Activity['type']) => {
    switch (type) {
      case 'deposit': return <ArrowDown className="h-4 w-4 text-green-500" />;
      case 'withdraw': return <ArrowUp className="h-4 w-4 text-red-500" />;
      case 'compound': return <RefreshCw className="h-4 w-4 text-mezo-primary" />;
    }
  };

  return (
    <Card>
      <CardHeader>
        <CardTitle>Recent Activity</CardTitle>
      </CardHeader>
      <CardContent>
        {activities.length === 0 ? (
          <p className="text-mezo-muted text-center py-4">No recent activity</p>
        ) : (
          <div className="space-y-3">
            {activities.map((act, i) => (
              <div key={i} className="flex items-center justify-between p-2 hover:bg-white/5 rounded-lg">
                <div className="flex items-center gap-3">
                  {getIcon(act.type)}
                  <div>
                    <p className="text-sm font-medium capitalize">{act.type}</p>
                    <p className="text-xs text-mezo-muted">
                      {act.user.slice(0, 6)}...{act.user.slice(-4)}
                    </p>
                  </div>
                </div>
                <p className="text-sm font-semibold">{Number(act.amount).toFixed(2)} MEZO</p>
              </div>
            ))}
          </div>
        )}
      </CardContent>
    </Card>
  );
}
```

## 5. Referral Widget

Encourages users to share their referral link and earn rewards.

```tsx
// components/referral/ReferralWidget.tsx
'use client';

import { useAccount } from 'wagmi';
import { useReferral } from '@/hooks/useReferral';
import { Card } from '@/components/ui/Card';
import { Button } from '@/components/ui/Button';
import { Copy, Check, Users } from 'lucide-react';
import { useState } from 'react';

export function ReferralWidget() {
  const { address } = useAccount();
  const { referralCode, totalRewards, claimRewards } = useReferral();
  const [copied, setCopied] = useState(false);

  if (!address) return null;

  const referralLink = `${window.location.origin}?ref=${referralCode}`;

  const handleCopy = async () => {
    await navigator.clipboard.writeText(referralLink);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <Card className="p-4">
      <div className="flex items-center gap-2 mb-3">
        <Users className="h-5 w-5 text-mezo-primary" />
        <h3 className="font-semibold">Refer & Earn</h3>
      </div>
      {referralCode ? (
        <>
          <div className="flex items-center gap-2 bg-mezo-darker p-2 rounded-lg mb-3">
            <code className="text-xs flex-1 truncate">{referralLink}</code>
            <Button variant="outline" size="sm" onClick={handleCopy}>
              {copied ? <Check className="h-4 w-4" /> : <Copy className="h-4 w-4" />}
            </Button>
          </div>
          <p className="text-sm text-mezo-muted">
            Earn 5% of every referred user's deposit value.
          </p>
          {totalRewards > 0 && (
            <Button onClick={claimRewards} className="mt-3 w-full">
              Claim {totalRewards} MEZO
            </Button>
          )}
        </>
      ) : (
        <p className="text-sm text-mezo-muted">Set your referral code in the vault to start earning.</p>
      )}
    </Card>
  );
}
```

### 6.2. Error Boundary for Contract Calls

```tsx
// components/ErrorBoundary.tsx
'use client';

import { Component, ErrorInfo, ReactNode } from 'react';
import { Button } from '@/components/ui/Button';
import { AlertTriangle } from 'lucide-react';

interface Props { children: ReactNode; }
interface State { hasError: boolean; error?: Error; }

export class ContractErrorBoundary extends Component<Props, State> {
  constructor(props: Props) { super(props); this.state = { hasError: false }; }
  static getDerivedStateFromError(error: Error) { return { hasError: true, error }; }
  componentDidCatch(error: Error, errorInfo: ErrorInfo) { console.error('Contract error:', error, errorInfo); }

  render() {
    if (this.state.hasError) {
      return (
        <div className="p-8 text-center">
          <AlertTriangle className="h-12 w-12 text-yellow-500 mx-auto mb-4" />
          <h3 className="text-lg font-semibold mb-2">Something went wrong</h3>
          <p className="text-mezo-muted mb-4">{this.state.error?.message}</p>
          <Button onClick={() => this.setState({ hasError: false })}>Try Again</Button>
        </div>
      );
    }
    return this.props.children;
  }
}
```

## 7. Custom Hooks for Data Fetching

### 7.1. useVaultStats Hook

Combines on‑chain and subgraph data.

```typescript
// hooks/useVaultStats.ts
import { useReadContract, useReadContracts } from 'wagmi';
import { CONTRACTS, VeMEZOVaultABI } from '@/lib/contracts';
import { formatEther } from 'viem';

export function useVaultStats() {
  const { data, isLoading } = useReadContracts({
    contracts: [
      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'totalUnderlying' },
      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'getPendingRewards' },
    ],
  });

  const totalUnderlying = data?.[0]?.result as bigint | undefined;
  const pending = data?.[1]?.result as bigint | undefined;

  const tvl = totalUnderlying ? Number(formatEther(totalUnderlying)) : 0;
  const projectedAPR = tvl > 0 ? 50 + Math.min(30, tvl / 100000) : 0; // example calculation
  const pendingRewards = pending ? Number(formatEther(pending)) : 0;

  return { tvl, projectedAPR, pendingRewards, isLoading };
}
```

```typescript
// hooks/useUserPosition.ts
import { useAccount, useReadContracts } from 'wagmi';
import { CONTRACTS, VeMEZOVaultABI } from '@/lib/contracts';
import { formatEther } from 'viem';

export function useUserPosition() {
  const { address } = useAccount();

  const { data } = useReadContracts({
    contracts: [
      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'balanceOf', args: [address!] },
      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'getUserTokenIds', args: [address!] },
    ],
    query: { enabled: !!address },
  });

  const shares = data?.[0]?.result as bigint | undefined;
  const tokenIds = (data?.[1]?.result as bigint[]) || [];

  const { data: totalSupply } = useReadContract({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    functionName: 'totalSupply',
  });

  const sharePercentage = shares && totalSupply
    ? (Number(formatEther(shares)) / Number(formatEther(totalSupply))) * 100
    : 0;

  const { data: totalUnderlying } = useReadContract({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    functionName: 'totalUnderlying',
  });

  const underlyingValue = shares && totalUnderlying && totalSupply
    ? (Number(formatEther(shares)) / Number(formatEther(totalSupply))) * Number(formatEther(totalUnderlying))
    : 0;

  return {
    shares: shares ? formatEther(shares) : '0',
    sharePercentage: sharePercentage.toFixed(2),
    underlyingValue: underlyingValue.toFixed(2),
    depositedTokenIds: tokenIds,
  };
}
```

### 7.3. useReferral Hook

```typescript
// hooks/useReferral.ts
import { useReadContract, useWriteContract } from 'wagmi';
import { CONTRACTS, ReferralManagerABI } from '@/lib/contracts';
import { useAccount } from 'wagmi';

export function useReferral() {
  const { address } = useAccount();

  const { data: referralCode } = useReadContract({
    address: CONTRACTS.REFERRAL_MANAGER,
    abi: ReferralManagerABI,
    functionName: 'referralCodeOf',
    args: address ? [address] : undefined,
  });

  const { data: totalRewards } = useReadContract({
    address: CONTRACTS.REFERRAL_MANAGER,
    abi: ReferralManagerABI,
    functionName: 'totalReferralRewards',
    args: address ? [address] : undefined,
  });

  const { writeContract: claimRewards } = useWriteContract();
  const handleClaim = () => {
    claimRewards({
      address: CONTRACTS.REFERRAL_MANAGER,
      abi: ReferralManagerABI,
      functionName: 'claimReferralRewards',
    });
  };

  return {
    referralCode: referralCode as string || '',
    totalRewards: totalRewards ? Number(totalRewards) : 0,
    claimRewards: handleClaim,
  };
}
```
