// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

interface IMorpho {
    function supply(address asset, uint256 amount, address onBehalf, bytes calldata params) external;
    function borrow(address asset, uint256 amount, uint256 interestRateMode, address onBehalf, address receiver, bytes calldata params) external;
    function withdraw(address asset, uint256 amount, address to, address owner) external;
}

interface IUpshiftVault {
    function deposit(uint256 assets, address receiver) external returns (uint256 shares);
    function redeem(uint256 shares, address receiver, address owner) external returns (uint256 assets);
}

contract Looper is Ownable, ReentrancyGuard {
    IERC20 public immutable musd;
    IMorpho public immutable morpho;
    IUpshiftVault public immutable upshiftVault;
    
    mapping(address => Position) public userPositions;
    
    struct Position {
        uint256 collateral;     // upMUSD collateral
        uint256 debt;           // Borrowed MUSD
        uint256 loops;          // Loop iterations
        uint256 healthFactor;   // Liquidation safety
        bool active;
    }
    
    uint256 public constant MAX_LOOPS = 5;
    uint256 public constant MIN_HEALTH_FACTOR = 1500; // 1.5x
    
    event LoopCreated(address indexed user, uint256 loops, uint256 collateral);
    event LoopUnwound(address indexed user, uint256 recovered);
    
    constructor(address _musd, address _morpho, address _upshift) {
        musd = IERC20(_musd);
        morpho = IMorpho(_morpho);
        upshiftVault = IUpshiftVault(_upshift);
    }
    
    function createLoop(uint256 initialDeposit, uint256 maxLoops) external nonReentrant {
        require(maxLoops <= MAX_LOOPS, "Max loops exceeded");
        require(initialDeposit >= 1000e6, "Min deposit"); // $1k MUSD
        
        musd.transferFrom(msg.sender, address(this), initialDeposit);
        
        uint256 collateral = initialDeposit;
        uint256 debt;
        
        // Loop: deposit → borrow → repeat
        for (uint256 i = 0; i < maxLoops; i++) {
            // 1. Deposit MUSD → get upMUSD
            uint256 shares = upshiftVault.deposit(collateral, address(this));
            
            // 2. Use upMUSD as collateral to borrow MUSD
            uint256 borrowAmount = _calculateBorrowAmount(shares, i);
            morpho.borrow(address(musd), borrowAmount, 2, address(this), address(this), "");
            
            collateral = borrowAmount; // Borrowed MUSD becomes next deposit
            debt += borrowAmount;
        }
        
        // Store position
        userPositions[msg.sender] = Position({
            collateral: initialDeposit + debt, // Total position size
            debt: debt,
            loops: maxLoops,
            healthFactor: _getHealthFactor(msg.sender),
            active: true
        });
        
        emit LoopCreated(msg.sender, maxLoops, collateral);
    }
    
    function unwindLoop() external nonReentrant {
        Position storage pos = userPositions[msg.sender];
        require(pos.active, "No active loop");
        
        // Repay all debt first
        musd.approve(address(morpho), pos.debt);
        morpho.repay(address(musd), pos.debt, 2, address(this));
        
        // Withdraw all collateral from vault
        upshiftVault.redeem(pos.collateral, msg.sender, msg.sender);
        
        pos.active = false;
        emit LoopUnwound(msg.sender, pos.collateral);
    }
    
    function _calculateBorrowAmount(uint256 collateralShares, uint256 iteration) internal pure returns (uint256) {
        // Conservative borrowing: 70% LTV, decreasing per loop
        uint256 ltv = 7000 - (iteration * 500); // 70% → 55%
        return (collateralShares * ltv) / 10000;
    }
    
    function _getHealthFactor(address user) internal view returns (uint256) {
        // Query Morpho for real health factor
        return 1800; // Simplified 1.8x
    }
    
    function emergencyUnwind(address user) external onlyOwner {
        // Emergency liquidation protection
        Position storage pos = userPositions[user];
        unwindLoopForUser(user);
    }
}

hooks/useLoopSimulator.ts

"use client";

import { useState, useCallback } from "react";

export function useLoopSimulator() {
  const [simulation, setSimulation] = useState({
    initialDeposit: 10000,
    loops: 3,
    apy: 0,
    healthFactor: 0,
    liquidationPrice: 0,
  });

  const runSimulation = useCallback(() => {
    // Simulate loop economics
    const ltvPerLoop = 0.7;
    const borrowRate = 0.01; // 1%
    const supplyApy = 0.30;   // 30%
    
    const totalLTV = simulation.loops * ltvPerLoop;
    const netApy = (supplyApy * totalLTV) - borrowRate;
    const healthFactor = 1 / (totalLTV * 0.8); // Conservative
    
    setSimulation({
      initialDeposit: 10000,
      loops: 3,
      apy: netApy * 100,
      healthFactor,
      liquidationPrice: (1 - 1/healthFactor) * 100,
    });
  }, []);

  return { simulation, runSimulation };
}
