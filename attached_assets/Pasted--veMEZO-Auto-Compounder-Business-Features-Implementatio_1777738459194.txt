// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "@openzeppelin/contracts/access/Ownable.sol";
import "./VeMEZOVault.sol";
import "./MultiStrategyVault.sol";

contract VaultFactory is Ownable {
    VeMEZOVault public immutable conservativeVault;
    MultiStrategyVault public immutable aggressiveVault;
    
    address public feeCollector;
    uint256 public constant PERFORMANCE_FEE = 1000; // 10% = 1000 bps
    
    event VaultDeployed(address indexed vault, string strategy);
    event FeeCollected(address indexed vault, uint256 amount);
    
    constructor(address _veMEZO, address _musd, address _feeCollector) {
        conservativeVault = new VeMEZOVault(
            _veMEZO, 
            _musd, 
            address(this), 
            PERFORMANCE_FEE,
            "Conservative"
        );
        aggressiveVault = new MultiStrategyVault(
            _veMEZO, 
            _musd, 
            address(this),
            PERFORMANCE_FEE
        );
        
        feeCollector = _feeCollector;
        emit VaultDeployed(address(conservativeVault), "Conservative");
        emit VaultDeployed(address(aggressiveVault), "Aggressive");
    }
    
    function collectFees(address[] calldata vaults) external onlyOwner {
        for (uint i = 0; i < vaults.length; i++) {
            uint256 fee = IVault(vaults[i]).collectPerformanceFee();
            IERC20(MUSD).transfer(feeCollector, fee);
            emit FeeCollected(vaults[i], fee);
        }
    }
}

interface IVault {
    function collectPerformanceFee() external returns (uint256);
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "@openzeppelin/contracts/token/ERC721/IERC721Receiver.sol";
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

interface IVeMEZO {
    function claimRebase(uint256 tokenId) external returns (uint256);
    function claimIncentives(uint256 tokenId) external returns (uint256);
    function increaseLockAmount(uint256 tokenId, uint256 amount) external;
}

contract VeMEZOVault is IERC721Receiver, ReentrancyGuard, Ownable {
    IVeMEZO public immutable veMEZO;
    IERC20 public immutable musd; // MUSD for fees
    address public immutable keeper;
    address public immutable feeCollector;
    
    uint256 public performanceFee; // bps (1000 = 10%)
    string public strategyName;
    
    mapping(uint256 => address) public nftOwner;
    uint256[] public depositedTokens;
    
    uint256 public totalUnderlying;
    uint256 public totalShares;
    
    event Deposited(address indexed user, uint256 tokenId, uint256 shares);
    event Withdrawn(address indexed user, uint256 shares, uint256 tokenId);
    event Compounded(uint256 totalRewards, uint256 fee);
    
    constructor(
        address _veMEZO,
        address _musd,
        address _keeper,
        uint256 _performanceFee,
        string memory _strategyName
    ) {
        veMEZO = IVeMEZO(_veMEZO);
        musd = IERC20(_musd);
        keeper = _keeper;
        performanceFee = _performanceFee;
        strategyName = _strategyName;
    }
    
    modifier onlyKeeper() {
        require(msg.sender == keeper, "Only keeper");
        _;
    }
    
    function deposit(uint256 tokenId) external nonReentrant {
        veMEZO.safeTransferFrom(msg.sender, address(this), tokenId);
        nftOwner[tokenId] = msg.sender;
        depositedTokens.push(tokenId);
        
        // Mint shares proportional to underlying value
        uint256 shares = _calculateShares(veMEZO.balanceOfNFT(tokenId));
        totalShares += shares;
        
        emit Deposited(msg.sender, tokenId, shares);
    }
    
    function compoundAll() external onlyKeeper nonReentrant {
        uint256 totalRewards = 0;
        uint256 totalFee = 0;
        
        for (uint i = 0; i < depositedTokens.length; i++) {
            uint256 tokenId = depositedTokens[i];
            if (nftOwner[tokenId] == address(0)) continue;
            
            uint256 rebase = veMEZO.claimRebase(tokenId);
            uint256 incentives = veMEZO.claimIncentives(tokenId);
            uint256 rewards = rebase + incentives;
            
            if (rewards > 0) {
                // Swap rewards to MEZO, take performance fee in MUSD
                uint256 fee = (rewards * performanceFee) / 10000;
                totalRewards += rewards - fee;
                totalFee += fee;
                
                // Relock rewards (simplified)
                veMEZO.increaseLockAmount(tokenId, rewards - fee);
            }
        }
        
        // Transfer fee to collector in MUSD equivalent
        musd.transfer(feeCollector, totalFee);
        emit Compounded(totalRewards, totalFee);
    }
    
    function onERC721Received(...) external pure override returns (bytes4) {
        return IERC721Receiver.onERC721Received.selector;
    }
    
    function _calculateShares(uint256 underlying) internal view returns (uint256) {
        if (totalUnderlying == 0 || totalShares == 0) {
            return underlying;
        }
        return (underlying * totalShares) / totalUnderlying;
    }
}
