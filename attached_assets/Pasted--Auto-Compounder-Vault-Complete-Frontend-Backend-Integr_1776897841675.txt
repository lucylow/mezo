# Auto‑Compounder Vault: Complete Frontend‑Backend Integration

This document provides a comprehensive, production‑ready implementation of the integration layer between the **Next.js frontend** and the **Mezo blockchain backend** for the veMEZO Auto‑Compounder Vault. It covers Web3 configuration with Mezo Passport, custom React hooks for contract interaction, subgraph data fetching, transaction management, state synchronization, and keeper bot communication. All code is written in TypeScript and follows best practices for type safety, error handling, and performance.

```typescript
import { CONTRACTS, VeMEZOVaultABI } from '@/lib/contracts';

export function useVaultStats() {
  const { data, isLoading, refetch } = useReadContracts({
    contracts: [

      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'totalSupply' },
      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'performanceFee' },
      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'getPendingRewards' },
    ],
  });

  const totalUnderlying = data?.[0]?.result as bigint | undefined;
  const totalSupply = data?.[1]?.result as bigint | undefined;
  const performanceFee = data?.[2]?.result as bigint | undefined;
  const pendingRewards = data?.[3]?.result as bigint | undefined;

  return {
    tvl: totalUnderlying ? Number(formatEther(totalUnderlying)) : 0,
    totalShares: totalSupply ? formatEther(totalSupply) : '0',
    performanceFee: performanceFee ? Number(performanceFee) / 100 : 10,
    pendingRewards: pendingRewards ? Number(formatEther(pendingRewards)) : 0,
    isLoading,
    refetch,
  };
}

export function useUserVaultPosition(address?: `0x${string}`) {
  const { data, isLoading, refetch } = useReadContracts({
    contracts: address ? [
      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'balanceOf', args: [address] },
      { address: CONTRACTS.VAULT, abi: VeMEZOVaultABI, functionName: 'getUserTokenIds', args: [address] },

    query: { enabled: !!address },
  });

    shares: shares ? formatEther(shares) : '0',
    tokenIds: tokenIds || [],

  };
```

```typescript
import { CONTRACTS, VeMEZOVaultABI, VeMEZOABI } from '@/lib/contracts';

export function useDeposit() {
  const queryClient = useQueryClient();
  const { writeContractAsync, data: hash, isPending } = useWriteContract();
  const { isLoading: isConfirming, isSuccess } = useWaitForTransactionReceipt({ hash });

  const deposit = async (tokenId: bigint) => {
    try {
      // 1. Approve veMEZO transfer
      const approveTx = await writeContractAsync({
        address: CONTRACTS.VEMEZO,
        abi: VeMEZOABI,
        functionName: 'approve',
        args: [CONTRACTS.VAULT, tokenId],
      });
      toast.loading('Approving veMEZO transfer...', { id: 'approve' });
      
      // Wait for approval (simplified - use waitForTransactionReceipt in production)
      
      // 2. Deposit
      const depositTx = await writeContractAsync({
        address: CONTRACTS.VAULT,
        abi: VeMEZOVaultABI,
        functionName: 'deposit',
        args: [tokenId],
      });
      
      toast.success('Deposit successful!', { id: 'deposit' });
      
      // Invalidate queries
      queryClient.invalidateQueries({ queryKey: ['vault'] });
      queryClient.invalidateQueries({ queryKey: ['user'] });
      
      return depositTx;
    } catch (error) {
      toast.error(`Deposit failed: ${error instanceof Error ? error.message : 'Unknown error'}`);
      throw error;
    }
  };

  return { deposit, isPending: isPending || isConfirming, isSuccess, hash };
}
```

### 2.3 useVeMEZOData – NFT and Balance Queries

```typescript
// hooks/contracts/useVeMEZOData.ts
import { useReadContracts } from 'wagmi';
import { formatEther } from 'viem';
import { CONTRACTS, VeMEZOABI } from '@/lib/contracts';

export interface VeMEZONFT {
  tokenId: bigint;
  value: bigint;
  lockAmount: bigint;
  lockEnd: Date;
}

export function useVeMEZONFTs(address?: `0x${string}`) {
  const { data: tokenIds } = useReadContract({
    address: CONTRACTS.VEMEZO,
    abi: VeMEZOABI,
    functionName: 'tokensOfOwner',
    args: address ? [address] : undefined,
    query: { enabled: !!address },
  });

  const tokenIdsArray = (tokenIds as bigint[]) || [];

  const { data: nftDetails, isLoading } = useReadContracts({
    contracts: tokenIdsArray.flatMap(tokenId => [
      { address: CONTRACTS.VEMEZO, abi: VeMEZOABI, functionName: 'balanceOfNFT', args: [tokenId] },
      { address: CONTRACTS.VEMEZO, abi: VeMEZOABI, functionName: 'locked', args: [tokenId] },
    ]),
    query: { enabled: tokenIdsArray.length > 0 },
  });

  const nfts: VeMEZONFT[] = [];
  for (let i = 0; i < tokenIdsArray.length; i++) {
    const value = nftDetails?.[i * 2]?.result as bigint | undefined;
    const locked = nftDetails?.[i * 2 + 1]?.result as [bigint, bigint] | undefined;
    if (value && locked) {
      nfts.push({
        tokenId: tokenIdsArray[i],
        value,
        lockAmount: locked[0],
        lockEnd: new Date(Number(locked[1]) * 1000),
      });
    }
  }

  return { nfts, isLoading };
}
```

## 3. Subgraph Integration

### 3.1 Subgraph Client

```typescript
// lib/subgraph/client.ts
const SUBGRAPH_URL = process.env.NEXT_PUBLIC_SUBGRAPH_URL || 
  'https://api.goldsky.com/api/public/project_clx.../subgraphs/vemezo-compounder/1.0.0/gn';

export async function subgraphRequest<T = any>(
  query: string,
  variables?: Record<string, any>
): Promise<T> {
  const response = await fetch(SUBGRAPH_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ query, variables }),
  });
  
  if (!response.ok) throw new Error(`Subgraph error: ${response.status}`);
  const json = await response.json();
  if (json.errors) throw new Error(json.errors[0].message);
  return json.data;
}
```

## 5. Keeper Bot Integration

### 5.1 Keeper API Route (Trigger Compounding)

```typescript
// app/api/keeper/compound/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { createPublicClient, http, createWalletClient } from 'viem';
import { privateKeyToAccount } from 'viem/accounts';
import { mezoTestnet } from '@/lib/constants/chains';
import { CONTRACTS, VeMEZOVaultABI } from '@/lib/contracts';

export async function POST(req: NextRequest) {
  const authHeader = req.headers.get('authorization');
  if (authHeader !== `Bearer ${process.env.KEEPER_API_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  try {
    const account = privateKeyToAccount(process.env.KEEPER_PRIVATE_KEY as `0x${string}`);
    const publicClient = createPublicClient({
      chain: mezoTestnet,
      transport: http(),
    });
    const walletClient = createWalletClient({
      account,
      chain: mezoTestnet,
      transport: http(),
    });

    // Check profitability
    const gasPrice = await publicClient.getGasPrice();
    const canCompound = await publicClient.readContract({
      address: CONTRACTS.VAULT,
      abi: VeMEZOVaultABI,
      functionName: 'checkUpkeep',
      args: [gasPrice],
    });

    if (!canCompound) {
      return NextResponse.json({ success: false, reason: 'not-profitable' });
    }

    const hash = await walletClient.writeContract({
      address: CONTRACTS.VAULT,
      abi: VeMEZOVaultABI,
      functionName: 'compoundAll',
    });

    return NextResponse.json({ success: true, txHash: hash });
  } catch (error) {
    return NextResponse.json({ success: false, error: (error as Error).message }, { status: 500 });
  }
}
```

### 7.1 Error Boundary for Contract Calls

```tsx
// components/error/ContractErrorBoundary.tsx
'use client';

import { Component, ErrorInfo, ReactNode } from 'react';
import { Button } from '@/components/ui/Button';
import { AlertTriangle } from 'lucide-react';

interface Props { children: ReactNode; fallback?: ReactNode; }
interface State { hasError: boolean; error?: Error; }

export class ContractErrorBoundary extends Component<Props, State> {
  constructor(props: Props) { super(props); this.state = { hasError: false }; }
  static getDerivedStateFromError(error: Error) { return { hasError: true, error }; }
  componentDidCatch(error: Error, errorInfo: ErrorInfo) { console.error('Contract error:', error, errorInfo); }

  render() {
    if (this.state.hasError) {
      return this.props.fallback || (
        <div className="p-6 text-center">
          <AlertTriangle className="h-12 w-12 text-yellow-500 mx-auto mb-4" />
          <h3 className="text-lg font-semibold mb-2">Contract Interaction Failed</h3>
          <p className="text-mezo-muted mb-4">{this.state.error?.message}</p>
          <Button onClick={() => this.setState({ hasError: false })}>Try Again</Button>
        </div>
      );
    }
    return this.props.children;
  }
}
```

### 7.2 Retry Utility for Failed Queries

```typescript
// lib/retry.ts
export async function withRetry<T>(
  fn: () => Promise<T>,
  options: { maxRetries?: number; delay?: number } = {}
): Promise<T> {
  const { maxRetries = 3, delay = 1000 } = options;
  let lastError: Error;
  
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (error) {
      lastError = error as Error;
      if (i < maxRetries - 1) {
        await new Promise(resolve => setTimeout(resolve, delay * Math.pow(2, i)));
      }
    }
  }
  throw lastError!;
}
```

```typescript
// hooks/useContractEvents.ts
import { useWatchContractEvent } from 'wagmi';
import { CONTRACTS, VeMEZOVaultABI } from '@/lib/contracts';
import { useTransactionStore } from '@/store/transactionStore';
import { formatEther } from 'viem';

export function useVaultEvents() {
  const addTransaction = useTransactionStore(s => s.addTransaction);

  useWatchContractEvent({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    eventName: 'Deposited',
    onLogs(logs) {
      logs.forEach(log => {
        const { user, value } = log.args;
        addTransaction({
          hash: log.transactionHash,
          type: 'deposit',
          status: 'confirmed',
          timestamp: Date.now(),
          amount: formatEther(value as bigint),
        });
      });
    },
  });

  useWatchContractEvent({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    eventName: 'Withdrawn',
    onLogs(logs) {
      logs.forEach(log => {
        const { user, value } = log.args;
        addTransaction({
          hash: log.transactionHash,
          type: 'withdraw',
          status: 'confirmed',
          timestamp: Date.now(),
          amount: formatEther(value as bigint),
        });
      });
    },
  });

  useWatchContractEvent({
    address: CONTRACTS.VAULT,
    abi: VeMEZOVaultABI,
    eventName: 'Compounded',
    onLogs(logs) {
      logs.forEach(log => {
        addTransaction({
          hash: log.transactionHash,
          type: 'compound',
          status: 'confirmed',
          timestamp: Date.now(),
        });
      });
    },
  });
}
```

## 9. Data Formatting & Utilities

### 9.1 Formatting Helpers

```typescript
// lib/utils/format.ts
import { formatUnits } from 'viem';

export function formatTokenAmount(amount: bigint, decimals: number = 18, displayDecimals: number = 4): string {
  const formatted = formatUnits(amount, decimals);
  const num = parseFloat(formatted);
  if (num === 0) return '0';
  if (num < 0.0001) return '<0.0001';
  return num.toFixed(displayDecimals);
}

export function formatNumber(num: number): string {
  if (num >= 1_000_000) return (num / 1_000_000).toFixed(2) + 'M';
  if (num >= 1_000) return (num / 1_000).toFixed(2) + 'K';
  return num.toFixed(2);
}

export function shortenAddress(addr: string): string {
  return `${addr.slice(0, 6)}...${addr.slice(-4)}`;
}
```

## 10. Putting It All Together: Example Page Component

```tsx
// app/vault/page.tsx
'use client';

import { useAccount } from 'wagmi';
import { useVaultStats, useUserVaultPosition } from '@/hooks/contracts/useVaultRead';
import { useDeposit, useWithdraw } from '@/hooks/contracts/useVaultWrite';
import { useVeMEZONFTs } from '@/hooks/contracts/useVeMEZOData';
import { useVaultEvents } from '@/hooks/useContractEvents';
import { useSyncVaultStore } from '@/hooks/useSyncVaultStore';
import { useTransactionToast } from '@/components/transactions/TransactionToast';
import { DepositForm } from '@/components/vault/DepositForm';
import { WithdrawForm } from '@/components/vault/WithdrawForm';
import { StatsOverview } from '@/components/vault/StatsOverview';
import { Card } from '@/components/ui/Card';

export default function VaultPage() {
  const { address, isConnected } = useAccount();
  
  // Sync store with on-chain data
  useSyncVaultStore();
  
  // Subscribe to real-time events
  useVaultEvents();
  
  const { tvl, projectedAPR, isLoading } = useVaultStats();
  const { shares, tokenIds } = useUserVaultPosition(address);
  const { nfts } = useVeMEZONFTs(address);
  const { deposit, isPending: isDepositing, hash: depositHash } = useDeposit();
  const { withdraw, isPending: isWithdrawing, hash: withdrawHash } = useWithdraw();
  
  // Transaction toasts
  useTransactionToast({ hash: depositHash, onSuccess: () => console.log('Deposit confirmed') });
  useTransactionToast({ hash: withdrawHash });

  if (!isConnected) {
    return <Card className="text-center py-12">Connect your wallet to manage veMEZO positions</Card>;
  }

  return (
    <div className="max-w-6xl mx-auto space-y-6 p-4">
      <StatsOverview tvl={tvl} apr={projectedAPR} isLoading={isLoading} />
      
      <div className="grid md:grid-cols-2 gap-6">
        <DepositForm 
          nfts={nfts.filter(n => !tokenIds.includes(n.tokenId))} 
          onDeposit={deposit} 
          isPending={isDepositing} 
        />
        <WithdrawForm 
          shares={shares} 
          tokenIds={tokenIds} 
          onWithdraw={withdraw} 
          isPending={isWithdrawing} 
        />
      </div>
    </div>
  );
}
```
