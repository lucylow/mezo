## 📋 Prompt Overview

**Repository:** `lucylow/mezo`  
**Branch:** `main` (or create feature branch `feat/decentralization`)  
**Goal:** Implement progressive decentralization features for the Auto‑Compounder Vault, aligning with Mezo's four‑pillar decentralization strategy: Supernormal Foundation oversight, Proof‑of‑HODL consensus, tBTC bridging, and ve(3,3) tokenomics.

**Context:** The vault currently operates with a single centralized keeper bot that triggers `compoundAll()`. // SPDX-License-Identifier: MIT
pragma solidity 0.8.26;

import "@gelatonetwork/automate/contracts/integrations/AutomateTaskCreator.sol";
import "../VeMEZOAutoCompounder.sol";

/**
 * @title GelatoCompounder
 * @dev Gelato integration for decentralized compounding automation
 * Implements IAutomateTaskCreator for Gelato compatibility
 */
contract GelatoCompounder is AutomateTaskCreator {
    VeMEZOAutoCompounder public immutable vault;
    
    // Gelato task ID for tracking
    bytes32 public taskId;
    
    // Minimum time between compounds (6 hours)
    uint256 public constant MIN_INTERVAL = 6 hours;
    uint256 public lastExecution;
    
    // Profitability threshold (basis points, 1000 = 10%)
    uint256 public minProfitMargin = 1000;
    
    event TaskCreated(bytes32 indexed taskId);
    event TaskCancelled(bytes32 indexed taskId);
    event CompoundedViaGelato(uint256 rewards, uint256 fee, uint256 compounded);
    
    constructor(address _vault) AutomateTaskCreator() {
        vault = VeMEZOAutoCompounder(_vault);
    }
    
    /**
     * @dev Gelato resolver: checks if compounding should execute
     */
    function checker()
        external
        view
        returns (bool canExec, bytes memory execPayload)
    {
        // Time check
        if (block.timestamp - lastExecution < MIN_INTERVAL) {
            return (false, bytes(""));
        }
        
        // Profitability check
        uint256 gasPrice = tx.gasprice;
        bool profitable = vault.checkUpkeep(gasPrice);
        
        if (!profitable) {
            return (false, bytes(""));
        }
        
        // Build execution payload
        execPayload = abi.encodeWithSelector(this.executeCompound.selector);
        return (true, execPayload);
    }
    
    /**
     * @dev Execution function called by Gelato
     */
    function executeCompound() external onlyDedicatedMsgSender {
        lastExecution = block.timestamp;
        
        (uint256 rewards, uint256 fee, uint256 compounded) = vault.compoundAll();
        
        emit CompoundedViaGelato(rewards, fee, compounded);
        
        // Pay Gelato fee (optional - can use vault's fee share)
        (uint256 feeDeduction, ) = _getFeeDetails();
        if (feeDeduction > 0) {
            _transfer(feeDeduction, dedicatedMsgSender);
        }
    }
    
    /**
     * @dev Create Gelato automation task
     */
    function createTask() external returns (bytes32) {
        require(taskId == bytes32(0), "Task already exists");
        
        bytes memory execData = abi.encodeWithSelector(this.executeCompound.selector);
        
        ModuleData memory moduleData = ModuleData({
            modules: new Module[](2),
            args: new bytes[](2)
        });
        
        moduleData.modules[0] = Module.RESOLVER;
        moduleData.modules[1] = Module.PROXY;
        moduleData.args[0] = abi.encode(address(this), this.checker.selector);
        moduleData.args[1] = bytes("");
        
        taskId = _createTask(address(this), execData, moduleData, address(0));
        
        emit TaskCreated(taskId);
        return taskId;
    }
    
    /**
     * @dev Cancel Gelato task
     */
    function cancelTask() external {
        require(taskId != bytes32(0), "No task exists");
        _cancelTask(taskId);
        emit TaskCancelled(taskId);
        taskId = bytes32(0);
    }
    
    /**
     * @dev Set minimum profit margin
     */
    function setMinProfitMargin(uint256 _margin) external {
        require(_margin <= 5000, "Max 50%");
        minProfitMargin = _margin;
    }
}

     * @dev Chainlink checkUpkeep
     */

        bytes calldata /* checkData */
    )
        external
        view
        override
        returns (bool upkeepNeeded, bytes memory performData)
    {
        // Time check
        if (block.timestamp - lastExecution < interval) {
            return (false, bytes(""));
        }
        
        // Profitability check
        uint256 gasPrice = tx.gasprice;
        bool profitable = vault.checkUpkeep(gasPrice);
        
        if (!profitable) {
            return (false, bytes(""));
        }
        
        performData = abi.encode(gasPrice);
        return (true, performData);
    }
    
    /**
     * @dev Chainlink performUpkeep
     */
    function performUpkeep(bytes calldata /* performData */) external override {
        lastExecution = block.timestamp;
        
        (uint256 rewards, uint256 fee, uint256 compounded) = vault.compoundAll();
        
        emit CompoundedViaChainlink(rewards, fee, compounded);
    }

// Add to existing VeMEZOAutoCompounder.sol

// New state variables
uint256 public feeDistributionRate = 5000; // 50% of fees to vveMEZO holders (basis points)
mapping(address => uint256) public pendingRewards;

event FeeDistributed(uint256 totalFee, uint256 toHolders, uint256 toTreasury);
event RewardsClaimed(address indexed user, uint256 amount);

/**
 * @dev Modified _collectFee to split between treasury and holders
 */
function _collectFee(uint256 feeAmount) internal returns (uint256 musdAmount) {
    if (feeAmount == 0) return 0;
    
    // Convert to MUSD (existing logic)
    musdAmount = _swapToMUSD(feeAmount);
    
    // Split fee
    uint256 toHolders = (musdAmount * feeDistributionRate) / 10000;
    uint256 toTreasury = musdAmount - toHolders;
    
    if (toHolders > 0) {
        _distributeToHolders(toHolders);
    }
    
    if (toTreasury > 0) {
        musdToken.safeTransfer(treasury, toTreasury);
    }
    
    emit FeeDistributed(musdAmount, toHolders, toTreasury);
    return musdAmount;
}

/**
 * @dev Distribute MUSD proportionally to vveMEZO holders
 */
function _distributeToHolders(uint256 amount) internal {
    uint256 totalShares = vaultToken.totalSupply();
    if (totalShares == 0) return;
    
    // Since we can't iterate all holders, we use a pull-based system
    // Add to a global pool that users can claim from
    feePool += amount;
    lastDistributionBlock = block.number;
    
    // Update accrual parameters for proportional claiming
    feePerShare = feePerShare + (amount * PRECISION) / totalShares;
}

/**
 * @dev Claim accumulated fee rewards for msg.sender
 */
function claimFeeRewards() external nonReentrant {
    uint256 userShares = vaultToken.balanceOf(msg.sender);
    require(userShares > 0, "No shares");
    
    uint256 pending = _calculatePendingRewards(msg.sender);
    require(pending > 0, "No pending rewards");
    
    // Reset user's reward debt
    userRewardDebt[msg.sender] = (userShares * feePerShare) / PRECISION;
    
    musdToken.safeTransfer(msg.sender, pending);
    
    emit RewardsClaimed(msg.sender, pending);
}

function _calculatePendingRewards(address user) internal view returns (uint256) {
    uint256 userShares = vaultToken.balanceOf(user);
    uint256 accumulated = (userShares * feePerShare) / PRECISION;
    uint256 debt = userRewardDebt[user];
    return accumulated > debt ? accumulated - debt : 0;
}

// SPDX-License-Identifier: MIT
pragma solidity 0.8.26;

import "@openzeppelin/contracts/access/AccessControl.sol";

/**
 * @title VaultMultiSig
 * @dev Multi‑signature access control for vault emergency functions
 */
contract VaultMultiSig is AccessControl {
    bytes32 public constant GUARDIAN_ROLE = keccak256("GUARDIAN_ROLE");
    bytes32 public constant EMERGENCY_ROLE = keccak256("EMERGENCY_ROLE");
    
    // Required signatures for emergency actions
    uint256 public requiredGuardians = 3;
    uint256 public requiredEmergency = 5;
    
    // Pending emergency action
    struct EmergencyAction {
        address target;
        bytes data;
        uint256 timestamp;
        mapping(address => bool) approvals;
        uint256 approvalCount;
        bool executed;
    }
    
    mapping(bytes32 => EmergencyAction) public emergencyActions;
    
    event EmergencyActionProposed(bytes32 indexed actionId, address target);
    event EmergencyActionApproved(bytes32 indexed actionId, address guardian);
    event EmergencyActionExecuted(bytes32 indexed actionId);
    
    constructor(address[] memory guardians, address[] memory emergencySigners) {
        _grantRole(DEFAULT_ADMIN_ROLE, msg.sender);
        
        for (uint256 i = 0; i < guardians.length; i++) {
            _grantRole(GUARDIAN_ROLE, guardians[i]);
        }
        for (uint256 i = 0; i < emergencySigners.length; i++) {
            _grantRole(EMERGENCY_ROLE, emergencySigners[i]);
        }
    }
    
    /**
     * @dev Propose an emergency action
     */
    function proposeEmergencyAction(
        address target,
        bytes calldata data
    ) external onlyRole(GUARDIAN_ROLE) returns (bytes32 actionId) {
        actionId = keccak256(abi.encodePacked(target, data, block.timestamp));
        
        EmergencyAction storage action = emergencyActions[actionId];
        action.target = target;
        action.data = data;
        action.timestamp = block.timestamp;
        
        emit EmergencyActionProposed(actionId, target);
    }
    
    /**
     * @dev Approve a pending emergency action
     */
    function approveEmergencyAction(bytes32 actionId) external onlyRole(GUARDIAN_ROLE) {
        EmergencyAction storage action = emergencyActions[actionId];
        require(!action.executed, "Already executed");
        require(!action.approvals[msg.sender], "Already approved");
        
        action.approvals[msg.sender] = true;
        action.approvalCount++;
        
        emit EmergencyActionApproved(actionId, msg.sender);
    }
    
    /**
     * @dev Execute approved emergency action
     */
    function executeEmergencyAction(bytes32 actionId) external onlyRole(GUARDIAN_ROLE) {
        EmergencyAction storage action = emergencyActions[actionId];
        require(!action.executed, "Already executed");
        require(action.approvalCount >= requiredGuardians, "Insufficient approvals");
        require(block.timestamp <= action.timestamp + 24 hours, "Action expired");
        
        action.executed = true;
        
        (bool success, ) = action.target.call(action.data);
        require(success, "Emergency action failed");
        
        emit EmergencyActionExecuted(actionId);
    }
}

  const votingDelay = 1 * 24 * 60 * 60 / 12; // 1 day in blocks
  const votingPeriod = 5 * 24 * 60 * 60 / 12; // 5 days
  const proposalThreshold = ethers.parseEther("10000");
  
  const Governor = await ethers.getContractFactory("VaultGovernor");
  const governor = await Governor.deploy(
    vaultTokenAddress,
    await timelock.getAddress(),
    votingDelay,
    votingPeriod,
    proposalThreshold
  );
  await governor.waitForDeployment();
  console.log("Governor deployed to:", await governor.getAddress());

  // 4. Deploy Gelato Compounder
  const GelatoCompounder = await ethers.getContractFactory("GelatoCompounder");
  const gelatoCompounder = await GelatoCompounder.deploy(await vault.getAddress());
  await gelatoCompounder.waitForDeployment();
  console.log("GelatoCompounder deployed to:", await gelatoCompounder.getAddress());

  // 5. Transfer vault ownership to Timelock
  await vault.transferOwnership(await timelock.getAddress());
  console.log("Vault ownership transferred to Timelock");

  // 6. Set up roles
  const PROPOSER_ROLE = await timelock.PROPOSER_ROLE();
  const EXECUTOR_ROLE = await timelock.EXECUTOR_ROLE();
  
  await timelock.grantRole(PROPOSER_ROLE, await governor.getAddress());
  await timelock.grantRole(EXECUTOR_ROLE, await governor.getAddress());
  console.log("Governor granted proposer/executor roles");

  console.log("\n=== Deployment Complete ===");
  console.log("Vault:", await vault.getAddress());
  console.log("Timelock:", await timelock.getAddress());
  console.log("Governor:", await governor.getAddress());
  console.log("GelatoCompounder:", await gelatoCompounder.getAddress());
}

main().catch(console.error);
