# Complete Wallet Connection Implementation for Auto‑Compounder Vault

## 1. Overview

This document provides a complete, production‑ready wallet connection system for the veMEZO Auto‑Compounder Vault. The implementation integrates **Mezo Passport**—a package built on RainbowKit that adds Bitcoin wallet support alongside standard EVM wallets—with **wagmi** and **viem** for robust wallet management across both ecosystems. The system handles all connection states, network switching, error recovery, and provides a polished user experience that satisfies the hackathon's Mezo Integration requirement (30% of judging).

The wallet connection architecture consists of the following layers:
1. **Configuration Layer**: Wagmi/Mezo Passport configuration with chain definitions and connectors
2. **Provider Layer**: React Context providers wrapping the entire application
3. **Hook Layer**: Custom hooks for connection state, account management, and balance queries
4. **Component Layer**: Connect button, account modal, network switcher, and error displays
5. **Utility Layer**: Helper functions for address formatting, balance conversion, and error handling

All code is written in TypeScript and follows best practices for type safety, error handling, and performance optimization.

## 3. Wagmi Configuration with Mezo Passport

The Mezo Passport package extends RainbowKit with Bitcoin wallet support, enabling users to connect with wallets like Xverse and Unisat alongside MetaMask and WalletConnect. It integrates with viem and wagmi for streamlined wallet management across both ecosystems.

### 3.1 `lib/wagmi/config.ts`

```typescript
import { http, createConfig, cookieStorage, createStorage } from 'wagmi';
import { mezoTestnet, mezoMainnet } from '@/lib/constants/chains';
import { getDefaultConfig } from '@rainbow-me/rainbowkit';
import { getMezoPassportConfig } from '@mezo-org/passport';

// Ensure WalletConnect Project ID is available
const projectId = process.env.NEXT_PUBLIC_WALLETCONNECT_PROJECT_ID;
if (!projectId) {
  throw new Error('NEXT_PUBLIC_WALLETCONNECT_PROJECT_ID is required');
}

// Create wagmi configuration with Mezo Passport integration
// getMezoPassportConfig() adds Bitcoin wallet support (Xverse, Unisat, etc.)
export const config = getDefaultConfig({
  appName: 'veMEZO Auto-Compounder',
  projectId,
  chains: [mezoTestnet, mezoMainnet],
  ssr: true,
  transports: {
    [mezoTestnet.id]: http(mezoTestnet.rpcUrls.default.http[0]),
    [mezoMainnet.id]: http(mezoMainnet.rpcUrls.default.http[0]),
  },
  // Enable cookie storage for SSR persistence
  storage: createStorage({
    storage: cookieStorage,
  }),
  ...getMezoPassportConfig(),
});

// Type for the config
export type Config = typeof config;

// Helper to get the current chain configuration
export function getChainConfig(chainId: number) {
  return [mezoTestnet, mezoMainnet].find((c) => c.id === chainId);
}
```

## 5. Custom Hooks for Wallet Management

### 5.1 `hooks/wallet/useWalletConnection.ts`

This is the primary hook for wallet connection state and actions.

```tsx
'use client';

import { useAccount, useConnect, useDisconnect, useSwitchChain, useChainId } from 'wagmi';
import { useCallback, useMemo } from 'react';
import { toast } from 'sonner';
import { SUPPORTED_CHAINS, DEFAULT_CHAIN } from '@/lib/constants/chains';
import { injected } from 'wagmi/connectors';

export interface WalletConnectionState {
  isConnected: boolean;
  isConnecting: boolean;
  isReconnecting: boolean;
  isSwitchingNetwork: boolean;
  address: `0x${string}` | undefined;
  chainId: number | undefined;
  chain: typeof SUPPORTED_CHAINS[number] | undefined;
  connector: string | undefined;
}

export interface WalletConnectionActions {
  connect: (connectorId?: string) => Promise<void>;
  connectInjected: () => Promise<void>;
  disconnect: () => Promise<void>;
  switchNetwork: (chainId: number) => Promise<void>;
  switchToDefaultChain: () => Promise<void>;
}

export function useWalletConnection(): WalletConnectionState & WalletConnectionActions {
  const { address, isConnected, chainId, chain, connector } = useAccount();
  const { connectAsync, connectors, isPending: isConnecting } = useConnect();
  const { disconnectAsync } = useDisconnect();
  const { switchChainAsync, isPending: isSwitchingNetwork } = useSwitchChain();

  const connect = useCallback(
    async (connectorId?: string) => {
      try {
        const targetConnector = connectorId
          ? connectors.find((c) => c.id === connectorId)
          : connectors[0];
        
        if (!targetConnector) {
          throw new Error('No wallet connector available');
        }

        await connectAsync({ connector: targetConnector });
        toast.success('Wallet connected successfully');
      } catch (error) {
        const errorMessage = error instanceof Error ? error.message : 'Failed to connect wallet';
        
        // Handle specific error types
        if (errorMessage.includes('User rejected')) {
          toast.error('Connection rejected by user');
        } else if (errorMessage.includes('Connector not found')) {
          toast.error('Wallet not installed. Please install a compatible wallet.');
        } else {
          toast.error(errorMessage);
        }
        
        throw error;
      }
    },
    [connectAsync, connectors]
  );

  const connectInjected = useCallback(async () => {
    const injectedConnector = connectors.find((c) => c.id === 'injected');
    if (injectedConnector) {
      await connect(injectedConnector.id);
    } else {
      toast.error('No injected wallet found. Please install MetaMask or another wallet.');
    }
  }, [connectors, connect]);

  const disconnect = useCallback(async () => {
    try {
      await disconnectAsync();
      toast.success('Wallet disconnected');
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Failed to disconnect';
      toast.error(errorMessage);
      throw error;
    }
  }, [disconnectAsync]);

  const switchNetwork = useCallback(
    async (targetChainId: number) => {
      try {
        await switchChainAsync({ chainId: targetChainId });
        const chainName = SUPPORTED_CHAINS.find((c) => c.id === targetChainId)?.name;
        toast.success(`Switched to ${chainName || 'network'}`);
      } catch (error) {
        const errorMessage = error instanceof Error ? error.message : 'Failed to switch network';
        
        if (errorMessage.includes('User rejected')) {
          toast.error('Network switch rejected');
        } else if (errorMessage.includes('Chain not configured')) {
          toast.error('Please add this network to your wallet');
        } else {
          toast.error(errorMessage);
        }
        
        throw error;
      }
    },
    [switchChainAsync]
  );

  const switchToDefaultChain = useCallback(async () => {
    await switchNetwork(DEFAULT_CHAIN.id);
  }, [switchNetwork]);

  const isReconnecting = useMemo(() => {
    return isConnecting && !isConnected;
  }, [isConnecting, isConnected]);

  return {
    // State
    isConnected,
    isConnecting,
    isReconnecting,
    isSwitchingNetwork,
    address,
    chainId,
    chain: chain as typeof SUPPORTED_CHAINS[number] | undefined,
    connector: connector?.name,
    // Actions
    connect,
    connectInjected,
    disconnect,
    switchNetwork,
    switchToDefaultChain,
  };
}
```

### 5.2 `hooks/wallet/useWalletBalances.ts`

Hook for fetching wallet balances on the Mezo network.

```tsx
'use client';

import { useAccount, useBalance, useReadContracts } from 'wagmi';
import { formatEther, formatUnits } from 'viem';
import { useMemo } from 'react';
import { CONTRACTS } from '@/lib/constants/addresses';

interface TokenBalance {
  symbol: string;
  name: string;
  balance: string;
  formatted: string;
  decimals: number;
}

interface WalletBalances {
  native: {
    symbol: string;
    balance: string;
    formatted: string;
  } | null;
  tokens: TokenBalance[];
  isLoading: boolean;
  refetch: () => void;
}

// Token configurations for the Mezo network
const TOKEN_CONFIGS = [
  {
    address: CONTRACTS.MEZO_TOKEN,
    symbol: 'MEZO',
    name: 'Mezo Token',
    decimals: 18,
  },
  {
    address: CONTRACTS.MUSD_TOKEN,
    symbol: 'MUSD',
    name: 'Mezo USD',
    decimals: 18,
  },
] as const;

export function useWalletBalances(): WalletBalances {
  const { address, isConnected } = useAccount();

  // Native BTC balance
  const {
    data: nativeBalance,
    isLoading: nativeLoading,
    refetch: refetchNative,
  } = useBalance({
    address,
    query: {
      enabled: isConnected,
      staleTime: 15_000,
    },
  });

  // ERC-20 token balances
  const {
    data: tokenBalances,
    isLoading: tokensLoading,
    refetch: refetchTokens,
  } = useReadContracts({
    contracts: TOKEN_CONFIGS.map((token) => ({
      address: token.address,
      abi: [
        {
          name: 'balanceOf',
          type: 'function',
          inputs: [{ name: 'account', type: 'address' }],
          outputs: [{ name: '', type: 'uint256' }],
          stateMutability: 'view',
        },
      ] as const,
      functionName: 'balanceOf',
      args: [address!],
    })),
    query: {
      enabled: isConnected && !!address,
      staleTime: 15_000,
    },
  });

  const tokens = useMemo(() => {
    if (!tokenBalances) return [];
    
    return TOKEN_CONFIGS.map((token, index) => {
      const balance = (tokenBalances[index]?.result as bigint) || 0n;
      return {
        symbol: token.symbol,
        name: token.name,
        balance: balance.toString(),
        formatted: formatUnits(balance, token.decimals),
        decimals: token.decimals,
      };
    });
  }, [tokenBalances]);

  const refetch = () => {
    refetchNative();
    refetchTokens();
  };

  return {
    native: nativeBalance
      ? {
          symbol: nativeBalance.symbol,
          balance: nativeBalance.value.toString(),
          formatted: nativeBalance.formatted,
        }
      : null,
    tokens,
    isLoading: nativeLoading || tokensLoading,
    refetch,
  };
}
```

### 5.3 `hooks/wallet/useNetworkStatus.ts`

Hook for monitoring network health and correct chain connection.

```tsx
'use client';

import { useAccount, useSwitchChain, useChainId } from 'wagmi';
import { useMemo } from 'react';
import { SUPPORTED_CHAINS, DEFAULT_CHAIN } from '@/lib/constants/chains';
import { toast } from 'sonner';

interface NetworkStatus {
  isCorrectNetwork: boolean;
  currentChain: typeof SUPPORTED_CHAINS[number] | undefined;
  requiredChain: typeof DEFAULT_CHAIN;
  isSwitching: boolean;
  switchToCorrectNetwork: () => Promise<void>;
  unsupportedNetwork: boolean;
}

export function useNetworkStatus(): NetworkStatus {
  const { isConnected } = useAccount();
  const chainId = useChainId();
  const { switchChainAsync, isPending: isSwitching } = useSwitchChain();

  const currentChain = useMemo(() => {
    if (!chainId) return undefined;
    return SUPPORTED_CHAINS.find((c) => c.id === chainId);
  }, [chainId]);

  const isCorrectNetwork = useMemo(() => {
    if (!isConnected) return true;
    return chainId === DEFAULT_CHAIN.id;
  }, [isConnected, chainId]);

  const unsupportedNetwork = useMemo(() => {
    if (!isConnected || !chainId) return false;
    return !SUPPORTED_CHAINS.some((c) => c.id === chainId);
  }, [isConnected, chainId]);

  const switchToCorrectNetwork = async () => {
    try {
      await switchChainAsync({ chainId: DEFAULT_CHAIN.id });
      toast.success(`Switched to ${DEFAULT_CHAIN.name}`);
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Failed to switch network';
      toast.error(errorMessage);
      throw error;
    }
  };

  return {
    isCorrectNetwork,
    currentChain,
    requiredChain: DEFAULT_CHAIN,
    isSwitching,
    switchToCorrectNetwork,
    unsupportedNetwork,
  };
}
```

### 5.4 `hooks/wallet/useWalletSigning.ts`

Hook for message signing and signature verification.

```tsx
'use client';

import { useAccount, useSignMessage, useSignTypedData } from 'wagmi';
import { useCallback } from 'react';
import { toast } from 'sonner';
import { hashMessage, verifyMessage, type HashType } from 'viem';

interface SigningResult {
  signature: string;
  signerAddress: string;
}

export function useWalletSigning() {
  const { address } = useAccount();
  const { signMessageAsync, isPending: isSigningMessage } = useSignMessage();
  const { signTypedDataAsync, isPending: isSigningTypedData } = useSignTypedData();

  const signMessage = useCallback(
    async (message: string): Promise<SigningResult | null> => {
      if (!address) {
        toast.error('Wallet not connected');
        return null;
      }

      try {
        const signature = await signMessageAsync({ message });
        return { signature, signerAddress: address };
      } catch (error) {
        const errorMessage = error instanceof Error ? error.message : 'Failed to sign message';
        
        if (errorMessage.includes('User rejected')) {
          toast.error('Signing cancelled');
        } else {
          toast.error(errorMessage);
        }
        
        return null;
      }
    },
    [address, signMessageAsync]
  );

  const signTypedData = useCallback(
    async (domain: any, types: any, value: any): Promise<SigningResult | null> => {
      if (!address) {
        toast.error('Wallet not connected');
        return null;
      }

      try {
        const signature = await signTypedDataAsync({
          domain,
          types,
          primaryType: Object.keys(types)[0],
          message: value,
        });
        return { signature, signerAddress: address };
      } catch (error) {
        const errorMessage = error instanceof Error ? error.message : 'Failed to sign typed data';
        toast.error(errorMessage);
        return null;
      }
    },
    [address, signTypedDataAsync]
  );

  const verifySignature = useCallback(
    (message: string, signature: string): boolean => {
      try {
        const recoveredAddress = verifyMessage({
          address: address!,
          message,
          signature: signature as `0x${string}`,
        });
        return recoveredAddress;
      } catch {
        return false;
      }
    },
    [address]
  );

  return {
    signMessage,
    signTypedData,
    verifySignature,
    isSigning: isSigningMessage || isSigningTypedData,
    isSigningMessage,
    isSigningTypedData,
  };
}
```

```tsx
'use client';

import { Component, ErrorInfo, ReactNode } from 'react';
import { Button } from '@/components/ui/Button';
import { AlertTriangle, RefreshCw } from 'lucide-react';
import { useWalletConnection } from '@/hooks/wallet/useWalletConnection';

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
  errorInfo: ErrorInfo | null;
}

export class ConnectionErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false, error: null, errorInfo: null };
  }

  static getDerivedStateFromError(error: Error): Partial<State> {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('Wallet Connection Error:', error, errorInfo);
    this.setState({ errorInfo });
  }

  render() {
    if (this.state.hasError) {
      return <ConnectionErrorFallback error={this.state.error} onReset={() => this.setState({ hasError: false })} />;
    }

    return this.props.children;
  }
}

function ConnectionErrorFallback({ error, onReset }: { error: Error | null; onReset: () => void }) {
  const { disconnect } = useWalletConnection();

  const handleReset = async () => {
    try {
```

### 7.2 Error Types & Utilities

`lib/wallet/errors.ts`

```typescript
export enum WalletErrorType {
  // Connection errors
  CONNECTOR_NOT_FOUND = 'ConnectorNotFoundError',
  CONNECTOR_ALREADY_CONNECTED = 'ConnectorAlreadyConnectedError',
  USER_REJECTED = 'UserRejectedRequestError',
  CHAIN_NOT_CONFIGURED = 'ChainNotConfiguredError',
  PROVIDER_NOT_FOUND = 'ProviderNotFoundError',
  
  // Network errors
  NETWORK_ERROR = 'NetworkError',
  CHAIN_MISMATCH = 'ChainMismatchError',
  UNSUPPORTED_CHAIN = 'UnsupportedChainError',
  
  // Transaction errors
  TRANSACTION_REJECTED = 'TransactionRejectedError',
  TRANSACTION_FAILED = 'TransactionFailedError',
  INSUFFICIENT_FUNDS = 'InsufficientFundsError',
  
  // General errors
  UNKNOWN_ERROR = 'UnknownError',
}

export interface WalletError {
  type: WalletErrorType;
  message: string;
  originalError?: Error;
}

export function parseWalletError(error: unknown): WalletError {
  const errorMessage = error instanceof Error ? error.message : String(error);
  const errorName = error instanceof Error ? error.name : 'UnknownError';
  
  // User rejection
  if (errorMessage.includes('User rejected') || errorMessage.includes('user rejected')) {
    return {
      type: WalletErrorType.USER_REJECTED,
      message: 'Transaction was rejected by user',
      originalError: error instanceof Error ? error : undefined,
    };
  }
  
  // Connector not found
  if (errorName === 'ConnectorNotFoundError' || errorMessage.includes('Connector not found')) {
    return {
      type: WalletErrorType.CONNECTOR_NOT_FOUND,
      message: 'Wallet connector not found. Please install a compatible wallet.',
      originalError: error instanceof Error ? error : undefined,
    };
  }
  
  // Chain mismatch
  if (errorMessage.includes('Chain mismatch') || errorName === 'ChainMismatchError') {
    return {
      type: WalletErrorType.CHAIN_MISMATCH,
      message: 'Please switch to the correct network in your wallet.',
      originalError: error instanceof Error ? error : undefined,
    };
  }
  
  // Unsupported chain
  if (errorName === 'UnsupportedChainError' || errorMessage.includes('Unsupported chain')) {
    return {
      type: WalletErrorType.UNSUPPORTED_CHAIN,
      message: 'This network is not supported. Please switch to Mezo.',
      originalError: error instanceof Error ? error : undefined,
    };
  }
  
  // Insufficient funds
  if (errorMessage.includes('insufficient funds') || errorMessage.includes('insufficient balance')) {
    return {
      type: WalletErrorType.INSUFFICIENT_FUNDS,
      message: 'Insufficient funds for transaction.',
      originalError: error instanceof Error ? error : undefined,
    };
  }
  
  // Default
  return {
    type: WalletErrorType.UNKNOWN_ERROR,
    message: errorMessage || 'An unknown error occurred',
    originalError: error instanceof Error ? error : undefined,
  };
}

export function getUserFriendlyErrorMessage(error: WalletError): string {
  switch (error.type) {
    case WalletErrorType.CONNECTOR_NOT_FOUND:
      return 'Wallet not detected. Please install MetaMask, Xverse, or another compatible wallet.';
    case WalletErrorType.USER_REJECTED:
      return 'Transaction cancelled. You can try again when ready.';
    case WalletErrorType.CHAIN_NOT_CONFIGURED:
    case WalletErrorType.CHAIN_MISMATCH:
      return 'Please switch to the Mezo network in your wallet.';
    case WalletErrorType.UNSUPPORTED_CHAIN:
      return 'This network is not supported. Please switch to Mezo.';
    case WalletErrorType.INSUFFICIENT_FUNDS:
      return 'You don\'t have enough BTC to cover the gas fees.';
    case WalletErrorType.TRANSACTION_FAILED:
      return 'Transaction failed. Please try again.';
    default:
      return error.message;
  }
}
```

## 8. Wallet Persistence & Session Management

### 8.1 `lib/wallet/persistence.ts`

```typescript
const STORAGE_KEYS = {
  LAST_CONNECTED_WALLET: 'vemezo:last_connected_wallet',
  LAST_CONNECTED_ADDRESS: 'vemezo:last_connected_address',
  PREFERRED_CHAIN: 'vemezo:preferred_chain',
} as const;

export interface WalletSession {
  connectorId: string;
  address: string;
  chainId: number;
  timestamp: number;
}

export function saveWalletSession(session: WalletSession): void {
  try {
    localStorage.setItem(STORAGE_KEYS.LAST_CONNECTED_WALLET, session.connectorId);
    localStorage.setItem(STORAGE_KEYS.LAST_CONNECTED_ADDRESS, session.address);
    localStorage.setItem(STORAGE_KEYS.PREFERRED_CHAIN, session.chainId.toString());
    localStorage.setItem(`${STORAGE_KEYS.LAST_CONNECTED_WALLET}:timestamp`, session.timestamp.toString());
  } catch (error) {
    console.warn('Failed to save wallet session:', error);
  }
}

export function getLastWalletSession(): WalletSession | null {
  try {
    const connectorId = localStorage.getItem(STORAGE_KEYS.LAST_CONNECTED_WALLET);
    const address = localStorage.getItem(STORAGE_KEYS.LAST_CONNECTED_ADDRESS);
    const chainId = localStorage.getItem(STORAGE_KEYS.PREFERRED_CHAIN);
    const timestamp = localStorage.getItem(`${STORAGE_KEYS.LAST_CONNECTED_WALLET}:timestamp`);
    
    if (!connectorId || !address || !chainId || !timestamp) {
      return null;
    }
    
    // Session expires after 30 days
    const sessionAge = Date.now() - parseInt(timestamp, 10);
    const MAX_SESSION_AGE = 30 * 24 * 60 * 60 * 1000;
    
    if (sessionAge > MAX_SESSION_AGE) {
      clearWalletSession();
      return null;
    }
    
    return {
      connectorId,
      address,
      chainId: parseInt(chainId, 10),
      timestamp: parseInt(timestamp, 10),
    };
  } catch (error) {
    console.warn('Failed to retrieve wallet session:', error);
    return null;
  }
}

export function clearWalletSession(): void {
  try {
    localStorage.removeItem(STORAGE_KEYS.LAST_CONNECTED_WALLET);
    localStorage.removeItem(STORAGE_KEYS.LAST_CONNECTED_ADDRESS);
    localStorage.removeItem(STORAGE_KEYS.PREFERRED_CHAIN);
    localStorage.removeItem(`${STORAGE_KEYS.LAST_CONNECTED_WALLET}:timestamp`);
  } catch (error) {
    console.warn('Failed to clear wallet session:', error);
  }
}

export function isSessionValid(session: WalletSession): boolean {
  const MAX_SESSION_AGE = 30 * 24 * 60 * 60 * 1000;
  return Date.now() - session.timestamp < MAX_SESSION_AGE;
}
```

```typescript
export function shortenAddress(address: string, chars = 4): string {
  if (!address) return '';
  if (address.length < chars * 2 + 2) return address;
  return `${address.slice(0, chars + 2)}...${address.slice(-chars)}`;
}

export function formatAddress(address: string): string {
  return shortenAddress(address, 4);
}

export function isValidAddress(address: string): boolean {
  return /^0x[a-fA-F0-9]{40}$/.test(address);
}

export function normalizeAddress(address: string): `0x${string}` {
  if (!address.startsWith('0x')) {
    return `0x${address}` as `0x${string}`;
  }
  return address as `0x${string}`;
}
```

```typescript
import { formatUnits } from 'viem';

export function formatNumber(num: number, decimals = 2): string {
  if (num >= 1_000_000_000) {
    return (num / 1_000_000_000).toFixed(decimals) + 'B';
  }
  if (num >= 1_000_000) {
    return (num / 1_000_000).toFixed(decimals) + 'M';
  }
  if (num >= 1_000) {
    return (num / 1_000).toFixed(decimals) + 'K';
  }
  return num.toFixed(decimals);
}

export function formatTokenAmount(amount: bigint, decimals: number, displayDecimals = 4): string {
  const formatted = formatUnits(amount, decimals);
  const num = parseFloat(formatted);
  
  if (num === 0) return '0';
  if (num < 0.0001) return '<0.0001';
  
  return num.toFixed(displayDecimals);
}

export function formatUSD(amount: number): string {
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD',
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(amount);
}
```
