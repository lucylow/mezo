"use client";

import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { WagmiProvider } from "wagmi";
import { RainbowKitProvider, getDefaultConfig } from "@rainbow-me/rainbowkit";
import { getMezoPassportConfig, mezoMatsnetTestnet } from "mezo-orgpassport";
import { useState } from "react";

const config = getDefaultConfig({
  appName: "veMEZO Auto-Compounder",
  projectId: process.env.NEXT_PUBLIC_WALLETCONNECT_PROJECT_ID!,
  chains: [mezoMatsnetTestnet, ...getMezoPassportConfig().chains],
});

export function Providers({ children }: { children: React.ReactNode }) {
  const [queryClient] = useState(() => new QueryClient());
  return (
    <WagmiProvider config={config}>
      <QueryClientProvider client={queryClient}>
        <RainbowKitProvider>{children}</RainbowKitProvider>
      </QueryClientProvider>
    </WagmiProvider>
  );
}

"use client";

import { useAccount, useSwitchChain, useDisconnect } from "wagmi";
import { ConnectButton } from "@rainbow-me/rainbowkit";
import { ChevronDown, Bitcoin, AlertCircle } from "lucide-react";
import { useState } from "react";

export function WalletConnectButton() {
  const { address, isConnected } = useAccount();
  const { disconnect } = useDisconnect();
  const [menuOpen, setMenuOpen] = useState(false);

  if (!isConnected) {
    return (
      <ConnectButton.Custom>
        {({ account, chain, openConnectModal, mounted }) => {
          return (
            <div>
              <button onClick={openConnectModal} className="px-6 py-3 bg-mezo-primary text-black font-semibold rounded-xl hover:opacity-90">
                Connect Wallet
              </button>
            </div>
          );
        }}
      </ConnectButton.Custom>
    );
  }

  return (
    <div className="relative">
      <button
        onClick={() => setMenuOpen(!menuOpen)}
        className="flex items-center gap-2 px-4 py-2 rounded-xl border border-border hover:bg-white/5 transition"
      >
        <Bitcoin className="h-4 w-4 text-mezo-primary" />
        <span className="text-sm font-medium">
          {address?.slice(0, 6)}...{address?.slice(-4)}
        </span>
        <ChevronDown className="h-4 w-4 text-mezo-muted" />
      </button>

      {menuOpen && (
        <div className="absolute right-0 mt-2 w-56 bg-mezo-darker border border-border rounded-2xl shadow-xl z-50">
          <div className="p-4 border-b border-border">
            <p className="text-xs text-mezo-muted">Bitcoin Wallet</p>
            <p className="text-sm font-mono">{address?.slice(0, 12)}...{address?.slice(-8)}</p>

            Disconnect

import { cn } from "@/lib/utils";

interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  loading?: boolean;
  fullWidth?: boolean;
  variant?: "primary" | "secondary";
  iconRight?: ReactNode;
}

export function Button({
  children,
  className,
  loading,
  fullWidth,
  variant = "primary",
  iconRight,
  ...props
}: ButtonProps) {
  return (
    <button
      disabled={loading}
      className={cn(
        "px-6 py-3 font-semibold rounded-xl transition-all disabled:opacity-50 disabled:cursor-not-allowed",
        variant === "primary" && "bg-gradient-to-r from-mezo-primary to-mezo-secondary text-black hover:opacity-90 shadow-lg hover:shadow-glow",
        variant === "secondary" && "bg-white/5 border border-border hover:bg-white/10",
        fullWidth && "w-full",
        className
      )}
      {...props}
    >
      {loading ? "Processing..." : children}
      {iconRight}
    </button>
  );
}

    abi: VAULT_ABI,
    functionName: "totalUnderlying",
  });

  const { data: totalSupply, isLoading: loadingSupply } = useReadContract({
    address: VAULT_ADDRESS,
    abi: VAULT_ABI,
    functionName: "totalSupply",
  });

  const tvl = totalUnderlying ? Number(formatEther(totalUnderlying)) : 0;
  const apr = 80; // Derived from 7-day historical

  return {
    stats: { tvl: Math.round(tvl), apr },
    isLoading: loadingTVL || loadingSupply,
  };
}
