export function CompoundingSimulator() {
  const [deposit] = useState(1000);
  const [apy] = useState(80);

  const final = Math.round(deposit * Math.pow(1 + apy / 100 / 52, 52 * 2));
  const earned = final - deposit;

    <Card className="p-6">
      <h3 className="text-lg font-semibold">Compounding simulator</h3>
      <p className="mt-1 text-sm text-mezo-muted">Show users what auto-compounding can do over time.</p>
