```tsx
  item: NavItem;
  isActive: boolean;

        <>

            {item.title}
          </span>
          {item.badge && (
            <span className="ml-auto bg-mezo-primary/20 text-mezo-primary text-xs px-2 py-0.5 rounded-full">
              {item.badge}
            </span>
          )}
        </>
      )}
    </>
  );

  const className = cn(
    "flex items-center w-full px-3 py-2 rounded-lg transition-colors",
    isActive ? "bg-mezo-primary/10" : "hover:bg-white/5",
    collapsed && "justify-center"
  );

  if (item.external) {
    return (
      <a
        href={item.href}
        target="_blank"
        rel="noopener noreferrer"
        className={className}
      >
        {linkContent}
      </a>
    );
  }

  return (
    <Link href={item.href} className={className}>
      {linkContent}
    </Link>
  );
}
```

### 2.4 Breadcrumb Navigation

`components/navigation/Breadcrumb.tsx`

```tsx
'use client';

import { usePathname } from 'next/navigation';
import Link from 'next/link';
import { ChevronRight, Home } from 'lucide-react';
import { Fragment } from 'react';

const routeLabels: Record<string, string> = {
  '': 'Dashboard',
  'vault': 'Vault',
  'portfolio': 'Portfolio',
  'earn': 'Earn',
  'swap': 'Swap',
  'governance': 'Governance',
  'analytics': 'Analytics',
  'leaderboard': 'Leaderboard',
  'history': 'History',
  'docs': 'Documentation',
  'settings': 'Settings',
  'community': 'Community',
};

export function Breadcrumb() {
  const pathname = usePathname();
  const segments = pathname.split('/').filter(Boolean);

  if (segments.length === 0) {
    return (
      <div className="flex items-center text-sm">
        <Home size={16} className="mr-2" />
        <span>Dashboard</span>
      </div>
    );
  }

  return (
    <nav className="flex items-center text-sm">
      <Link href="/" className="text-mezo-muted hover:text-white transition">
        <Home size={16} />
      </Link>
      {segments.map((segment, index) => {
        const href = '/' + segments.slice(0, index + 1).join('/');
        const label = routeLabels[segment] || segment;
        const isLast = index === segments.length - 1;

        return (
          <Fragment key={href}>
            <ChevronRight size={14} className="mx-2 text-mezo-muted" />
            {isLast ? (
              <span className="text-white font-medium">{label}</span>
            ) : (
              <Link href={href} className="text-mezo-muted hover:text-white transition">
                {label}
              </Link>
            )}
          </Fragment>
        );
      })}
    </nav>
  );
}
```

### 2.6 Search Modal

`components/navigation/SearchModal.tsx`

```tsx
'use client';

import { Fragment, useState, useEffect } from 'react';
import { Dialog, Transition } from '@headlessui/react';
import { Search, FileText, Vault, TrendingUp, ArrowLeftRight } from 'lucide-react';
import { useRouter } from 'next/navigation';
import { cn } from '@/lib/utils';

interface SearchResult {
  title: string;
  href: string;
  icon: React.ElementType;
  description?: string;
}

const searchData: SearchResult[] = [
  { title: 'Dashboard', href: '/', icon: Vault, description: 'View your vault overview' },
  { title: 'Vault', href: '/vault', icon: Vault, description: 'Deposit and withdraw veMEZO' },
  { title: 'Portfolio', href: '/portfolio', icon: TrendingUp, description: 'Track your positions' },
  { title: 'Earn', href: '/earn', icon: TrendingUp, description: 'Explore yield strategies' },
  { title: 'Swap', href: '/swap', icon: ArrowLeftRight, description: 'Swap MEZO and MUSD' },
  { title: 'Documentation', href: '/docs', icon: FileText, description: 'Learn about the protocol' },
];

interface SearchModalProps {
  open: boolean;
  onClose: () => void;
}

export function SearchModal({ open, onClose }: SearchModalProps) {
  const router = useRouter();
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<SearchResult[]>([]);

  useEffect(() => {
    if (!query) {
      setResults([]);
      return;
    }
    const filtered = searchData.filter(item =>
      item.title.toLowerCase().includes(query.toLowerCase()) ||
      item.description?.toLowerCase().includes(query.toLowerCase())
    );
    setResults(filtered);
  }, [query]);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'k' && (e.metaKey || e.ctrlKey)) {
        e.preventDefault();
        open ? onClose() : onClose(); // This would need open state from parent
      }
    };
    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [open, onClose]);

  const handleSelect = (href: string) => {
    router.push(href);
    onClose();
  };

  return (
    <Transition appear show={open} as={Fragment}>
      <Dialog as="div" className="relative z-50" onClose={onClose}>
        <Transition.Child
          as={Fragment}
          enter="ease-out duration-300"
          enterFrom="opacity-0"
          enterTo="opacity-100"
          leave="ease-in duration-200"
          leaveFrom="opacity-100"
          leaveTo="opacity-0"
        >
          <div className="fixed inset-0 bg-black/80 backdrop-blur-sm" />
        </Transition.Child>

        <div className="fixed inset-0 overflow-y-auto">
          <div className="flex min-h-full items-start justify-center p-4 pt-20">
            <Transition.Child
              as={Fragment}
              enter="ease-out duration-300"
              enterFrom="opacity-0 scale-95"
              enterTo="opacity-100 scale-100"
              leave="ease-in duration-200"
              leaveFrom="opacity-100 scale-100"
              leaveTo="opacity-0 scale-95"
            >
              <Dialog.Panel className="w-full max-w-xl transform overflow-hidden rounded-2xl bg-mezo-darker border border-border shadow-xl transition-all">
                <div className="flex items-center border-b border-border px-4">
                  <Search className="h-5 w-5 text-mezo-muted" />
                  <input
                    type="text"
                    className="w-full bg-transparent px-3 py-4 text-white placeholder-mezo-muted focus:outline-none"
                    placeholder="Search pages, commands..."
                    value={query}
                    onChange={(e) => setQuery(e.target.value)}
                    autoFocus
                  />
                </div>
                <div className="max-h-96 overflow-y-auto p-2">
                  {results.length === 0 && query && (
                    <p className="text-mezo-muted text-center py-8">No results found</p>
                  )}
                  {results.length === 0 && !query && (
                    <div className="py-4">
                      <p className="text-xs text-mezo-muted px-2 mb-2">Quick Navigation</p>
                      {searchData.slice(0, 5).map((item) => (
                        <SearchResultItem key={item.href} item={item} onSelect={handleSelect} />
                      ))}
                    </div>
                  )}
                  {results.map((item) => (
                    <SearchResultItem key={item.href} item={item} onSelect={handleSelect} />
                  ))}
                </div>
                <div className="border-t border-border px-4 py-3 flex items-center justify-between text-xs text-mezo-muted">
                  <span>Press ↵ to select</span>
                  <span>ESC to close</span>
                </div>
              </Dialog.Panel>
            </Transition.Child>
          </div>
        </div>
      </Dialog>
    </Transition>
  );
}

function SearchResultItem({ item, onSelect }: { item: SearchResult; onSelect: (href: string) => void }) {
  const Icon = item.icon;
  return (
    <button
      onClick={() => onSelect(item.href)}
      className="flex items-center w-full px-3 py-2 rounded-lg hover:bg-white/5 transition text-left"
    >
      <Icon className="h-5 w-5 text-mezo-muted mr-3" />
      <div>
        <p className="text-sm font-medium">{item.title}</p>
        {item.description && (
          <p className="text-xs text-mezo-muted">{item.description}</p>
        )}
      </div>
    </button>
  );
}
```

### 2.7 Navigation Events (Loading Indicator)

`components/navigation/NavigationEvents.tsx`

```tsx
'use client';

import { useEffect, useState } from 'react';
import { usePathname, useSearchParams } from 'next/navigation';
import { LoadingBar } from '@/components/ui/LoadingBar';

export function NavigationEvents() {
  const pathname = usePathname();
  const searchParams = useSearchParams();
  const [isNavigating, setIsNavigating] = useState(false);

  useEffect(() => {
    setIsNavigating(true);
    const timeout = setTimeout(() => setIsNavigating(false), 500);
    return () => clearTimeout(timeout);
  }, [pathname, searchParams]);

  return <LoadingBar loading={isNavigating} />;
}
```

### 16.2 Copy Address Button

`components/ui/CopyButton.tsx`

```tsx
'use client';

import { useState } from 'react';
import { Copy, Check } from 'lucide-react';
import { cn } from '@/lib/utils';

interface CopyButtonProps {
  text: string;
  className?: string;
}

export function CopyButton({ text, className }: CopyButtonProps) {
  const [copied, setCopied] = useState(false);

  const handleCopy = async () => {
    await navigator.clipboard.writeText(text);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <button
      onClick={handleCopy}
      className={cn("p-1 hover:bg-white/5 rounded transition", className)}
    >
      {copied ? <Check size={14} className="text-mezo-accent" /> : <Copy size={14} />}
    </button>
  );
}
```
